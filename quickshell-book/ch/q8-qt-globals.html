XMLHttpRequest example#

function fetchJson(url, cb) {
    const xhr = new XMLHttpRequest()
    xhr.onreadystatechange = () => {
        if (xhr.readyState === XMLHttpRequest.DONE)
            cb(xhr.status === 200 ? JSON.parse(xhr.responseText) : null)
    }
    xhr.open("GET", url)
    xhr.send()
}
