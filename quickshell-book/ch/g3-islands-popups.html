g3Gallery: Dynamic Islands, Popups and OSDs

Floating elements that appear, morph and disappear: the Dynamic Island, toasts, on-screen displays, cheatsheets. These are their own windows, so put them directly inside ShellRoot { }, not inside the bar.

Dynamic Island#

What it is. An iPhone-style black pill at the top center of the screen. It shows the time when idle, grows into a strip with album art and moving bars while music plays, briefly widens with the volume level when you change it, pops out notifications, and expands into full media controls when you click it. Every change is a spring-like morph.

Kind: its own window · Needs: PipeWire, MPRIS (optional: your notification service)

galleryshell/popups/DynamicIsland.qmlimport Quickshell
import Quickshell.Wayland
import Quickshell.Widgets
import Quickshell.Services.Mpris
import Quickshell.Services.Pipewire
import QtQuick
import QtQuick.Layouts

Scope {
    id: root

    // ---------- state ----------
    readonly property MprisPlayer player: Mpris.players.values.find(p => p.isPlaying) ?? null
    readonly property PwNode sink: Pipewire.defaultAudioSink
    PwObjectTracker { objects: [root.sink] }

    property bool expanded: false          // clicked open
    property string flash: ""              // "volume" or "notify": temporary states
    property string notifyTitle: ""
    property string notifyBody: ""
    function showFlash(kind: string): void { flash = kind; flashTimer.restart() }
    Timer { id: flashTimer; interval: 2000; onTriggered: root.flash = "" }

    // volume changes → flash the volume state (ignore the first value at startup)
    property bool armed: false
    Timer { interval: 1500; running: true; onTriggered: root.armed = true }
    Connections {
        target: root.sink?.audio ?? null
        function onVolumeChanged() { if (root.armed) root.showFlash("volume") }
        function onMutedChanged() { if (root.armed) root.showFlash("volume") }
    }
    // call this from your notification service: island.notify(n.summary, n.body)
    function notify(title: string, body: string): void { notifyTitle = title; notifyBody = body; showFlash("notify") }

    // which mode wins right now
    readonly property string mode: expanded && player ? "expanded"
                                 : flash === "notify" ? "notify"
                                 : flash === "volume" ? "volume"
                                 : player ? "media" : "idle"
    readonly property size targetSize: ({
        idle:     Qt.size(120, 32),
        media:    Qt.size(300, 40),
        volume:   Qt.size(260, 40),
        notify:   Qt.size(380, 72),
        expanded: Qt.size(400, 140)
    })[mode]

    PanelWindow {
        id: win
        anchors.top: true
        margins.top: 6
        implicitWidth: 420
        implicitHeight: 160
        exclusionMode: ExclusionMode.Ignore
        color: "transparent"
        WlrLayershell.layer: WlrLayer.Overlay
        WlrLayershell.namespace: "quickshell:island"
        mask: Region { item: island }                 // only the pill itself catches clicks

        Rectangle {
            id: island
            anchors.horizontalCenter: parent.horizontalCenter
            y: 0
            width: root.targetSize.width
            height: root.targetSize.height
            radius: Math.min(height / 2, 28)
            color: "#000000"
            clip: true
            // springy morph between sizes
            Behavior on width  { SpringAnimation { spring: 4; damping: 0.32; epsilon: 0.5 } }
            Behavior on height { SpringAnimation { spring: 4; damping: 0.32; epsilon: 0.5 } }

            TapHandler { onTapped: root.expanded = !root.expanded }

            // ---- idle: clock ----
            SystemClock { id: clock; precision: SystemClock.Minutes }
            Text {
                anchors.centerIn: parent
                opacity: root.mode === "idle" ? 1 : 0
                Behavior on opacity { NumberAnimation { duration: 150 } }
                text: Qt.formatDateTime(clock.date, "hh:mm")
                color: "white"; font.pixelSize: 13; font.bold: true
            }

            RowLayout {
                anchors { fill: parent; leftMargin: 6; rightMargin: 14 }
                opacity: root.mode === "media" ? 1 : 0
                visible: opacity > 0
                Behavior on opacity { NumberAnimation { duration: 180 } }
                spacing: 10
                ClippingRectangle {
                    implicitWidth: 28; implicitHeight: 28; radius: 8; color: "#313244"
                    Image { anchors.fill: parent; source: root.player?.trackArtUrl ?? ""; fillMode: Image.PreserveAspectCrop; sourceSize: Qt.size(56, 56) }
                }
                Text {
                    Layout.fillWidth: true
                    text: (root.player?.trackTitle ?? "") + (root.player?.trackArtist ? " · " + root.player.trackArtist : "")
                    color: "white"; font.pixelSize: 12; elide: Text.ElideRight
                }
                Row {                                          // fake equalizer bars
                    spacing: 3
                    Repeater {
                        model: 4
                        Rectangle {
                            required property int index
                            width: 3; radius: 1.5; color: "#a6e3a1"
                            anchors.bottom: parent.bottom
                            height: 6
                            SequentialAnimation on height {
                                running: root.mode === "media"; loops: Animation.Infinite
                                NumberAnimation { to: 6 + (index % 2 ? 10 : 6); duration: 280 + index * 70 }
                                NumberAnimation { to: 5; duration: 260 + index * 50 }
                            }
                        }
                    }
                }
            }

            // ---- volume flash ----
            RowLayout {
                anchors { fill: parent; leftMargin: 16; rightMargin: 16 }
                opacity: root.mode === "volume" ? 1 : 0
                visible: opacity > 0
                Behavior on opacity { NumberAnimation { duration: 150 } }
                spacing: 10
                Text {
                    text: root.sink?.audio?.muted ? "volume_off" : "volume_up"
                    font.family: "Material Symbols Rounded"; font.pixelSize: 18; color: "white"
                }
                Rectangle {
                    Layout.fillWidth: true; implicitHeight: 6; radius: 3; color: "#45475a"
                    Rectangle {
                        width: parent.width * Math.min(1, root.sink?.audio?.muted ? 0 : (root.sink?.audio?.volume ?? 0))
                        height: parent.height; radius: 3; color: "white"
                        Behavior on width { NumberAnimation { duration: 120 } }
                    }
                }
            }

            // ---- notification ----
            ColumnLayout {
                anchors { fill: parent; margins: 14 }
                opacity: root.mode === "notify" ? 1 : 0
                visible: opacity > 0
                Behavior on opacity { NumberAnimation { duration: 180 } }
                spacing: 2
                Text { text: root.notifyTitle; color: "white"; font.bold: true; font.pixelSize: 13; elide: Text.ElideRight; Layout.fillWidth: true }
                Text { text: root.notifyBody; color: "#a6adc8"; font.pixelSize: 12; elide: Text.ElideRight; Layout.fillWidth: true }
            }

            // ---- expanded media controls ----
            RowLayout {
                anchors { fill: parent; margins: 16 }
                opacity: root.mode === "expanded" ? 1 : 0
                visible: opacity > 0
                Behavior on opacity { NumberAnimation { duration: 200 } }
                spacing: 14
                ClippingRectangle {
                    implicitWidth: 100; implicitHeight: 100; radius: 16; color: "#313244"
                    Image { anchors.fill: parent; source: root.player?.trackArtUrl ?? ""; fillMode: Image.PreserveAspectCrop; sourceSize: Qt.size(200, 200) }
                }
                ColumnLayout {
                    Layout.fillWidth: true
                    spacing: 4
                    Text { text: root.player?.trackTitle ?? ""; color: "white"; font.bold: true; font.pixelSize: 15; elide: Text.ElideRight; Layout.fillWidth: true }
                    Text { text: root.player?.trackArtist ?? ""; color: "#a6adc8"; font.pixelSize: 12; elide: Text.ElideRight; Layout.fillWidth: true }
                    Rectangle {                                  // progress
                        Layout.fillWidth: true; implicitHeight: 4; radius: 2; color: "#45475a"
                        Rectangle {
                            width: parent.width * ((root.player?.position ?? 0) / Math.max(1, root.player?.length ?? 1))
                            height: parent.height; radius: 2; color: "white"
                        }
                    }
                    RowLayout {
                        Layout.alignment: Qt.AlignHCenter
                        spacing: 22
                        Repeater {
                            model: [{ i: "skip_previous", f: () => root.player.previous() },
                                    { i: root.player?.isPlaying ? "pause" : "play_arrow", f: () => root.player.togglePlaying() },
                                    { i: "skip_next", f: () => root.player.next() }]
                            Text {
                                required property var modelData
                                text: modelData.i
                                font.family: "Material Symbols Rounded"; font.pixelSize: 26; color: "white"
                                TapHandler { onTapped: parent.modelData.f() }
                            }
                        }
                    }
                }
            }
        }
    }
    // keep the progress bar moving while expanded
    Timer { running: root.mode === "expanded" && (root.player?.isPlaying ?? false); interval: 1000; repeat: true
            onTriggered: root.player.positionChanged() }
}

Add to your shell: DynamicIsland { id: island }. From your notification service, call island.notify(n.summary, n.body).

Code walkthrough · piece by piece

This is the biggest widget in the gallery, but it's built from one simple idea: one black shape, one mode string, and content that fades in for each mode.

lines 1–7Imports
import Quickshell
import Quickshell.Wayland
import Quickshell.Widgets
import Quickshell.Services.Mpris
import Quickshell.Services.Pipewire
import QtQuick
import QtQuick.Layouts

Media comes from Quickshell.Services.Mpris (players like Spotify and browsers), volume from Pipewire, and ClippingRectangle from Quickshell.Widgets for rounded album art.

lines 9–15Data sources
Scope {
    id: root

    // ---------- state ----------
    readonly property MprisPlayer player: Mpris.players.values.find(p => p.isPlaying) ?? null
    readonly property PwNode sink: Pipewire.defaultAudioSink
    PwObjectTracker { objects: [root.sink] }

A Scope is an invisible container for non-visual things and windows. player is the first player that is currently playing (or null). sink is the default audio output, tracked so its volume is readable.

lines 17–22Temporary states
property bool expanded: false          // clicked open
property string flash: ""              // "volume" or "notify": temporary states
property string notifyTitle: ""
property string notifyBody: ""
function showFlash(kind: string): void { flash = kind; flashTimer.restart() }
Timer { id: flashTimer; interval: 2000; onTriggered: root.flash = "" }

expanded is the click-to-open state. flash holds a temporary mode ("volume" or "notify") that should disappear by itself. showFlash() sets it and restarts a 2-second timer; the timer clears it. Restarting means a second volume change while the first flash is showing extends it instead of cutting it short.

lines 24–31React to volume changes
// volume changes → flash the volume state (ignore the first value at startup)
property bool armed: false
Timer { interval: 1500; running: true; onTriggered: root.armed = true }
Connections {
    target: root.sink?.audio ?? null
    function onVolumeChanged() { if (root.armed) root.showFlash("volume") }
    function onMutedChanged() { if (root.armed) root.showFlash("volume") }
}

Connections listens to signals of the sink's audio object. The armed flag ignores changes during the first 1.5 seconds, because PipeWire reports the initial volume at startup and we don't want a flash then.

lines 32–33An API for notifications
// call this from your notification service: island.notify(n.summary, n.body)
function notify(title: string, body: string): void { notifyTitle = title; notifyBody = body; showFlash("notify") }

A public function your notification service can call: island.notify(n.summary, n.body). The island doesn't need to know where notifications come from.

lines 35–39Choosing the mode
// which mode wins right now
readonly property string mode: expanded && player ? "expanded"
                             : flash === "notify" ? "notify"
                             : flash === "volume" ? "volume"
                             : player ? "media" : "idle"

A chain of conditions with clear priorities: expanded beats a notification, which beats a volume flash, which beats media, which beats idle. Everything else in the widget just reads mode.

lines 40–46Size per mode
readonly property size targetSize: ({

The window is fixed at 420×160 (the biggest the island gets) and centered at the top. It doesn't push windows (ExclusionMode.Ignore) and lives on the Overlay layer above everything. mask: Region { item: island } is essential: only the black pill catches clicks; the transparent rest of the window lets clicks through to your apps.

lines 60–71The morphing shape
Rectangle {
    id: island
    anchors.horizontalCenter: parent.horizontalCenter
    y: 0
    width: root.targetSize.width
    height: root.targetSize.height

Width and height come from targetSize. radius: Math.min(height / 2, 28) keeps a pill shape when small and caps the roundness when tall. clip: true hides content outside the shape while it's still growing. SpringAnimation (instead of a normal easing) simulates a spring: spring is stiffness, damping how fast it calms down. Low damping (0.32) gives the slight wobble that makes it feel physical.

line 73Click to expand

Hold-to-Show Keybind Cheatsheet#

What it is. Hold a key and a grid of your keybinds appears; release it and it disappears. It reads your binds live from Hyprland, so it's always up to date.

Kind: its own window · Needs: Hyprland, jq

galleryshell/popups/Cheatsheet.qmlimport Quickshell
import Quickshell.Io
import Quickshell.Wayland
import Quickshell.Hyprland
import QtQuick
import QtQuick.Layouts

Scope {
    id: root
    property bool shown: false
    property var binds: []

    GlobalShortcut {
        name: "cheatsheet"
        onPressed: { loadBinds.running = true; root.shown = true }
        onReleased: root.shown = false
    }

    // hyprctl binds -j → [{ modmask, key, dispatcher, arg, description, … }]
    Process {
        id: loadBinds
        command: ["hyprctl", "binds", "-j"]
        stdout: StdioCollector {
            onStreamFinished: {
                const mods = m => [[64, "SUPER"], [1, "SHIFT"], [4, "CTRL"], [8, "ALT"]].filter(([b]) => m & b).map(([, n]) => n)
                root.binds = JSON.parse(text)
                    .filter(b => b.key && !b.mouse)
                    .map(b => ({ keys: [...mods(b.modmask), b.key].join("+"),
                                 what: b.description || (b.dispatcher + " " + b.arg).trim() }))
                    .slice(0, 40)
            }
        }
    }

    PanelWindow {
        implicitWidth: 760; implicitHeight: grid.implicitHeight + 80
        exclusionMode: ExclusionMode.Ignore
        color: "transparent"
        visible: root.shown || fade.running
        WlrLayershell.layer: WlrLayer.Overlay

        Rectangle {
            anchors.fill: parent
            radius: 22; color: "#ee1e1e2e"; border.color: "#45475a"
            opacity: root.shown ? 1 : 0
            Behavior on opacity { NumberAnimation { id: fade; duration: 150 } }

            Text { x: 24; y: 20; text: "Keybinds"; color: "#cba6f7"; font.pixelSize: 16; font.bold: true }
            GridLayout {
                id: grid
                anchors { left: parent.left; right: parent.right; top: parent.top; margins: 24; topMargin: 56 }
                columns: 2; columnSpacing: 24; rowSpacing: 6
                Repeater {
                    model: root.binds
                    RowLayout {
                        required property var modelData
                        Layout.fillWidth: true
                        spacing: 10
                        Rectangle {
                            implicitWidth: k.implicitWidth + 14; implicitHeight: 22; radius: 6; color: "#313244"
                            Text { id: k; anchors.centerIn: parent; text: modelData.keys; color: "#cdd6f4"; font.pixelSize: 11; font.family: "JetBrains Mono" }
                        }
                        Text { Layout.fillWidth: true; text: modelData.what; color: "#a6adc8"; font.pixelSize: 12; elide: Text.ElideRight }
                    }
                }
            }
        }
    }
}

hyprctl binds -j returns JSON. modmask is a bit field: each modifier is one bit (Super = 64, Shift = 1, Ctrl = 4, Alt = 8). mods() checks each bit with & and returns the names. Mouse binds and keyless entries are skipped. Each bind becomes { keys: "SUPER+SHIFT+Q", what: "…" }, using your bindd description when you wrote one. We keep 40 to fit the screen.
