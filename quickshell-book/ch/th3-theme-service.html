th3Theming 3: The Theme Service

The Theme singleton is the heart of the switcher. It loads the theme files, turns roles into colours, measures contrast, switches themes, remembers your choice, follows a day/night schedule and answers commands from the terminal. Every widget in the shell talks to it, and only to it.

This chapter walks through the whole file, every line, in order.

theming/services/Theme.qmlpragma Singleton
import Quickshell
import Quickshell.Io
import QtQuick
import qs.services

// The theme service. It loads every theme file, remembers your choice, and exposes the
// current theme's colours as properties. Widgets only ever use Theme.colors.<role>.
Singleton {
    id: root

    // ---- which theme is active ----
    property var themes: ({})                     // id -> theme object, filled by load()
    property string previewId: ""                 // set while hovering a card in the picker
    readonly property string chosenId: saved.autoSwitch ? scheduledId : saved.theme
    readonly property string activeId: previewId !== "" ? previewId : chosenId
    readonly property var theme: themes[activeId] ?? themes["catppuccin-mocha"] ?? fallback
    readonly property bool dark: theme.variant !== "light"

    // all themes, sorted by family, then name (for the picker)
    readonly property var list: Object.values(themes).sort((a, b) => a.family.localeCompare(b.family) || a.name.localeCompare(b.name))

    // ---- automatic day/night switching ----
    readonly property bool isDay: clock.hours >= saved.dayStart && clock.hours < saved.nightStart
    readonly property string scheduledId: isDay ? saved.dayTheme : saved.nightTheme

    // ---- the colours every widget uses ----
    readonly property QtObject colors: QtObject {
        readonly property color background: root.role("background")
        readonly property color surface: root.role("surface")
        readonly property color surfaceHigh: root.role("surfaceHigh")
        readonly property color border: root.role("border")
        readonly property color text: root.role("text")
        readonly property color textMuted: root.role("textMuted")
        readonly property color textFaint: root.role("textFaint")
        readonly property color primary: root.role("primary")
        readonly property color onPrimary: root.role("onPrimary")
        readonly property color secondary: root.role("secondary")
        readonly property color red: root.role("red")
        readonly property color orange: root.role("orange")
        readonly property color yellow: root.role("yellow")
        readonly property color green: root.role("green")
        readonly property color cyan: root.role("cyan")
        readonly property color blue: root.role("blue")
        readonly property color magenta: root.role("magenta")
        // meanings built from the accents
        readonly property color success: green
        readonly property color warning: yellow
        readonly property color error: red
    }

    // ---- looking up colours ----
    // A role's value is a palette name ("mauve") or a literal colour ("#cba6f7").
    function resolve(t: var, roleName: string): string {
        const v = t?.roles?.[roleName] ?? "";
        if (v.startsWith("#")) return v;
        return t?.palette?.[v] ?? "#ff00ff";          // bright pink = a mistake in the theme file
    }
    function role(roleName: string): color {
        return resolve(theme, roleName);
    }

    // ---- colour helpers ----
    function alpha(c: color, a: real): color {
        return Qt.rgba(c.r, c.g, c.b, a);
    }
    function mix(a: color, b: color, amount: real): color {      // amount 0 = a, 1 = b
        return Qt.rgba(a.r + (b.r - a.r) * amount, a.g + (b.g - a.g) * amount, a.b + (b.b - a.b) * amount, 1);
    }
    // WCAG contrast ratio between two colours: 1 (none) .. 21 (black on white). 4.5+ is good for text.
    function contrast(a: color, b: color): real {
        const lum = c => {
            const ch = x => x <= 0.03928 ? x / 12.92 : Math.pow((x + 0.055) / 1.055, 2.4);
            return 0.2126 * ch(c.r) + 0.7152 * ch(c.g) + 0.0722 * ch(c.b);
        };
        const l1 = lum(a), l2 = lum(b);
        return (Math.max(l1, l2) + 0.05) / (Math.min(l1, l2) + 0.05);
    }
    // the more readable of the theme's text and background colours on top of `bg`
    function readableOn(bg: color): color {
        return contrast(colors.text, bg) >= contrast(colors.background, bg) ? colors.text : colors.background;
    }

    // ---- switching ----
    function set(id: string): void {
        if (!themes[id]) return;
        previewId = "";
        saved.autoSwitch = false;
        saved.theme = id;
    }
    function step(delta: int): void {
        const ids = list.map(t => t.id);
        set(ids[(ids.indexOf(chosenId) + delta + ids.length) % ids.length]);
    }
    // jump to the same family's light or dark version (e.g. Mocha <-> Latte)
    function toggleVariant(): void {
        const wanted = dark ? "light" : "dark";
        const twin = list.find(t => t.family === theme.family && t.variant === wanted);
        if (twin) set(twin.id);
    }

    // ---- loading theme files ----
    // Built-in themes live in <config>/themes, your own in ~/.config/theming/themes.
    readonly property string userDir: (Quickshell.env("XDG_CONFIG_HOME") || Quickshell.env("HOME") + "/.config") + "/theming/themes"

    function load(): void {
        loader.running = true;
    }

    Component.onCompleted: load()

    Process {
        id: loader
        // print every theme file, separated by a line with a single ASCII "record separator"
        command: ["sh", "-c", 'for f in "$1"/*.json "$2"/*.json; do [ -f "$f" ] && cat "$f" && printf "\\n\\036\\n"; done',
                  "sh", Quickshell.shellPath("themes"), root.userDir]
        stdout: StdioCollector {
            onStreamFinished: {
                const found = {};
                for (const chunk of text.split("\u001e")) {
                    if (!chunk.trim()) continue;
                    try {
                        const t = JSON.parse(chunk);
                        if (t.id && t.palette && t.roles) found[t.id] = t;
                    } catch (e) {
                        console.warn("Theme: skipping a file that isn't valid JSON");
                    }
                }
                root.themes = found;               // user themes with the same id replace built-ins
            }
        }
    }

    // ---- remembering the choice ----
    FileView {
        path: Quickshell.statePath("theme.json")
        printErrors: false
        onAdapterUpdated: writeAdapter()
        onLoadFailed: writeAdapter()

        JsonAdapter {
            id: saved
            property string theme: "catppuccin-mocha"
            property bool autoSwitch: false
            property string dayTheme: "catppuccin-latte"
            property string nightTheme: "catppuccin-mocha"
            property int dayStart: 7
            property int nightStart: 19
        }
    }
    readonly property alias settings: saved

    SystemClock {
        id: clock
        precision: SystemClock.Minutes
    }

    // ---- shell commands: qs -c theming ipc call theme <function> ----
    IpcHandler {
        target: "theme"
        function set(id: string): void { root.set(id) }
        function next(): void { root.step(1) }
        function previous(): void { root.step(-1) }
        function toggleVariant(): void { root.toggleVariant() }
        function list(): string { return root.list.map(t => t.id).join("\n") }
        function get(): string { return root.chosenId }
        function reload(): void { root.load() }
        function auto(on: bool): void { saved.autoSwitch = on }
    }

    // used before the theme files have loaded (Catppuccin Mocha)
    readonly property var fallback: ({
        id: "fallback", name: "Fallback", family: "Fallback", variant: "dark",
        palette: {},
        roles: { background: "#1e1e2e", surface: "#181825", surfaceHigh: "#313244", border: "#45475a", text: "#cdd6f4",
                 textMuted: "#a6adc8", textFaint: "#6c7086", primary: "#cba6f7", onPrimary: "#1e1e2e", secondary: "#89b4fa",
                 red: "#f38ba8", orange: "#fab387", yellow: "#f9e2af", green: "#a6e3a1", cyan: "#94e2d5", blue: "#89b4fa", magenta: "#f5c2e7" }
    })
}

Line by line#

Code walkthrough · piece by piece
line 1A singleton
pragma Singleton

pragma Singleton makes exactly one Theme for the whole shell. Any file that writes import qs.services can use Theme.colors.primary.

lines 2–5Imports
import Quickshell
import Quickshell.Io
import QtQuick
import qs.services

Quickshell for Singleton, SystemClock and Quickshell.shellPath; Quickshell.Io for Process, FileView, JsonAdapter and IpcHandler; QtQuick for the basics. qs.services is the file's own folder: importing it explicitly keeps tools like qmllint happy about other singletons there.

line 13All loaded themes
property var themes: ({})                     // id -> theme object, filled by load()

An object from id to theme, like { "nord": {…}, "dracula": {…} }. It starts empty and is filled by load() at the bottom. var because it's plain JavaScript data.

line 14The preview
property string previewId: ""                 // set while hovering a card in the picker

While you hover a card in the picker, its id goes here and the whole shell shows that theme. It's never saved.

line 15The chosen theme
readonly property string chosenId: saved.autoSwitch ? scheduledId : saved.theme

What you picked, or, when automatic switching is on, the theme the schedule says (scheduledId, line 25). Because this is a binding, flipping autoSwitch immediately changes the chosen theme.

line 16The active theme
readonly property string activeId: previewId !== "" ? previewId : chosenId

The preview if there is one, otherwise the chosen theme. Everything visible follows activeId; saving and exporting follow chosenId. Keeping the two apart is what makes previews safe.

line 17The theme object
readonly property var theme: themes[activeId] ?? themes["catppuccin-mocha"] ?? fallback

Look up the active id. If it doesn't exist (a typo, or the files haven't loaded yet), use Catppuccin Mocha, and if even that's missing, the built-in fallback at the end of the file. The ?? operator means "if the left side is null or undefined, use the right side".

line 18Dark or light
readonly property bool dark: theme.variant !== "light"

A convenience: true unless the theme says "variant": "light".

line 21The sorted list
readonly property var list: Object.values(themes).sort((a, b) => a.family.localeCompare(b.family) || a.name.localeCompare(b.name))

All themes as an array, sorted by family, then by name. localeCompare sorts text properly (accents included). The picker and next/previous use this order.

lines 24–25Day or night
readonly property bool isDay: clock.hours >= saved.dayStart && clock.hours < saved.nightStart
readonly property string scheduledId: isDay ? saved.dayTheme : saved.nightTheme

isDay is true between dayStart and nightStart (7:00 and 19:00 by default). scheduledId picks the day or night theme. clock is the SystemClock near the bottom; with minute precision, the binding re-checks every minute, so the switch happens by itself at 7:00 and 19:00.

lines 54–58Resolving a role
function resolve(t: var, roleName: string): string {
    const v = t?.roles?.[roleName] ?? "";
    if (v.startsWith("#")) return v;
    return t?.palette?.[v] ?? "#ff00ff";          // bright pink = a mistake in the theme file
}

Take the role's value from the theme. If it starts with #, it's already a colour. Otherwise it's a palette name, so look it up in the palette. If anything is missing, return #ff00ff (bright pink) so a broken theme file is easy to spot. The ?. operator means "stop and give undefined if the left side is null", so a half-written theme can't crash the shell.

lines 59–61role()

    return resolve(theme, roleName);
}

resolve for the active theme. Declaring the return type color converts the hex string into a real colour.

lines 64–66alpha()

The same colour with a different opacity: Theme.alpha(Theme.colors.primary, 0.2) for a soft tint.

lines 67–69mix()

    return Qt.rgba(a.r + (b.r - a.r) * amount, a.g + (b.g - a.g) * amount, a.b + (b.b - a.b) * amount, 1);

Blends two colours: mix(background, primary, 0.1) is the background with 10% of the accent, handy for subtle hover colours that suit every theme.

lines 71–78contrast()
function contrast(a: color, b: color): real {
    const lum = c => {
        const ch = x => x <= 0.03928 ? x / 12.92 : Math.pow((x + 0.055) / 1.055, 2.4);
        return 0.2126 * ch(c.r) + 0.7152 * ch(c.g) + 0.0722 * ch(c.b);
    };

    return (Math.max(l1, l2) + 0.05) / (Math.min(l1, l2) + 0.05);

The WCAG contrast ratio from Theming 1: Colours, Palettes and Roles. lum computes a colour's relative luminance: each channel is converted from screen values to light intensity (the ch function), then weighted by how sensitive our eyes are to red, green and blue (green counts most). The ratio is (lighter + 0.05) / (darker + 0.05).

lines 80–82readableOn()
function readableOn(bg: color): color {
    return contrast(colors.text, bg) >= contrast(colors.background, bg) ? colors.text : colors.background;
}

For a background you don't control (an album cover's colour, a tag colour), pick whichever of the theme's text or background colour is more readable on it.

lines 85–90set()
function set(id: string): void {
    if (!themes[id]) return;
    previewId = "";
    saved.autoSwitch = false;
    saved.theme = id;
}

Choose a theme for good: ignore unknown ids, end any preview, turn off automatic switching (you've made a manual choice), and store the id in the saved settings, which writes it to disk.

lines 91–94step()
function step(delta: int): void {
    const ids = list.map(t => t.id);
    set(ids[(ids.indexOf(chosenId) + delta + ids.length) % ids.length]);
}

Next (+1) or previous (−1) theme in the sorted list, wrapping around at both ends. Adding ids.length before % keeps the index positive when going back from the first theme.

lines 96–100toggleVariant()
function toggleVariant(): void {

    if (twin) set(twin.id);

Find a theme with the same family and the opposite variant (Mocha → Latte, Tokyo Night → Tokyo Night Day) and switch to it. Families without a twin (Nord, Dracula) simply don't change.

line 104Your themes folder
readonly property string userDir: (Quickshell.env("XDG_CONFIG_HOME") || Quickshell.env("HOME") + "/.config") + "/theming/themes"

~/.config/theming/themes (respecting XDG_CONFIG_HOME if set). Themes you write or download go here, outside the kit's own folder, so updating the kit never deletes them.

lines 106–110Loading
function load(): void {
    loader.running = true;
}

Component.onCompleted: load()

load() starts the loader process; it runs once at startup and again when you call reload over IPC.

lines 112–116Reading every file at once
Process {
    id: loader
    // print every theme file, separated by a line with a single ASCII "record separator"
    command: ["sh", "-c", 'for f in "$1"/*.json "$2"/*.json; do [ -f "$f" ] && cat "$f" && printf "\\n\\036\\n"; done',

One small shell loop prints every .json file from the built-in folder (Quickshell.shellPath("themes") is themes/ inside this config) and then from your folder. After each file it prints a line containing the ASCII "record separator" character (code 30, written \036), which never appears in JSON, so the output can be split back into files. Both folder paths are passed as arguments ($1, $2), never pasted into the command, so unusual folder names can't break anything.

lines 117–131Parsing
stdout: StdioCollector {
    onStreamFinished: {
        const found = {};
        for (const chunk of text.split("\u001e")) {
            if (!chunk.trim()) continue;
            try {
                const t = JSON.parse(chunk);
                if (t.id && t.palette && t.roles) found[t.id] = t;

Split the output on the separator, parse each piece, and keep it if it has an id, a palette and roles. A broken file is skipped with a warning instead of breaking the whole switcher. Your folder is read last, so a file of yours with a built-in's id replaces the built-in. Assigning root.themes once at the end updates the list and the picker in one step.

lines 135–150Saving your choice

A FileView + JsonAdapter pair stores six settings in the shell's state folder (Quickshell.statePath, under ~/.local/state/quickshell/): the theme, whether automatic switching is on, the day and night themes, and the two switch hours. onAdapterUpdated: writeAdapter() saves every change; onLoadFailed: writeAdapter() creates the file on the first run.

Trying it from a terminal#

qs -c theming ipc call theme list            # every theme id
qs -c theming ipc call theme set tokyo-night
qs -c theming ipc call theme toggleVariant   # Tokyo Night -> Tokyo Night Day
qs -c theming ipc call theme next
qs -c theming ipc call theme auto true       # follow the day/night schedule

Change the schedule by editing ~/.local/state/quickshell/by-shell/<id>/theme.json (or from the exercises below).

Compared to Caelestia and end-4

Caelestia's Colours service (Caelestia 2: Tokens, Colours and Building Blocks) and end-4's MaterialThemeLoader (end-4 2: Appearance, Config and Common Widgets) load a single generated scheme (from the wallpaper or a preset) into Material roles. This service keeps a library of named themes and switches between them; Theming 7: Theming Your Whole Desktop adds a wallpaper-generated theme to the same library.
