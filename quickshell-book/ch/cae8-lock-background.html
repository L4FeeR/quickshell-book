cae8Caelestia 8: Lock Screen, Wallpaper and Shortcuts

Three pieces live outside the drawers window: the lock screen (a real Wayland session lock), the background (wallpaper and desktop clock on the bottom layer), and the shortcuts daemon that connects keybinds to drawers. This chapter also covers the wallpaper service that re-themes the shell.

modules/background/  Background
services/            Wallpapers  Time
daemons/             Shortcuts

Module 1: The lock controller#

cae/modules/lock/Lock.qmlimport Quickshell
import Quickshell.Io
import Quickshell.Wayland
import Quickshell.Services.Pam
import QtQuick

// The lock screen controller. WlSessionLock asks the compositor for a real lock: if the shell
// crashes, the session stays locked. One LockSurface is created per monitor.
Scope {
    id: root

    property bool locked: false
    property string password: ""
    property bool busy: false
    property string message: ""
    signal failed()

    function submit(): void {
        if (busy || password === "") return;
        busy = true;
        message = "";
        pam.start();
    }

    PamContext {
        id: pam
        onPamMessage: if (responseRequired) respond(root.password)
        onCompleted: result => {
            root.busy = false;
            if (result === PamResult.Success) {
                root.locked = false;
            } else {
                root.message = "Wrong password";
                root.failed();
            }
            root.password = "";
        }
    }

    WlSessionLock {
        locked: root.locked
        WlSessionLockSurface {
            LockSurface { lock: root }
        }
    }

    IpcHandler {
        target: "lock"
        function lock(): void { root.locked = true }
        function isLocked(): bool { return root.locked }
    }
}

lines 12–16Shared state
property bool locked: false
property string password: ""
property bool busy: false
property string message: ""
signal failed()

The password lives here, not in the surfaces, because there's one surface per monitor and they all share the same field. failed is a signal so every surface can shake its input at once.

lines 18–23Submitting
function submit(): void {
    if (busy || password === "") return;
    busy = true;
    message = "";
    pam.start();
}

Ignore Enter while a check is running or when nothing was typed; otherwise start PAM.

lines 25–38PAM
PamContext {
    id: pam
    onPamMessage: if (responseRequired) respond(root.password)
    onCompleted: result => {
        root.busy = false;
        if (result === PamResult.Success) {
            root.locked = false;
        } else {
            root.message = "Wrong password";
            root.failed();
        }
        root.password = "";
    }
}

PamContext talks to the system's authentication (the same thing sudo and your login use; it also handles fingerprint readers). When PAM asks a question that needs an answer (responseRequired), we answer with the password. On success the lock is released; otherwise a message is shown and failed fires. The password is always cleared.

lines 40–45The real lock
WlSessionLock {
    locked: root.locked
    WlSessionLockSurface {
        LockSurface { lock: root }
    }
}

WlSessionLock uses the Wayland session-lock protocol. The compositor itself keeps the screen covered until the lock is released, even if Quickshell crashes, so it's safe. WlSessionLockSurface creates one surface per monitor, each containing a LockSurface.

lines 47–51IPC
IpcHandler {
    target: "lock"
    function lock(): void { root.locked = true }
    function isLocked(): bool { return root.locked }
}

qs -c cae ipc call lock lock locks the screen (bind it to a key or run it from hypridle). There's deliberately no IPC to unlock.

Test with a safety net

Before trusting any lock screen, add a temporary Timer { running: root.locked; interval: 60000; onTriggered: root.locked = false } to Lock.qml, test wrong and right passwords, then remove the timer. If something goes wrong anyway, switch to a TTY (Ctrl+Alt+F2) and run pkill qs; the compositor then shows its own fallback until you unlock with loginctl unlock-session.

Singleton {

One clock for the whole shell. format() wraps Qt.formatDateTime, so modules write Time.format("hh:mm").

Module 4: Wallpapers and theming#

cae/services/Wallpapers.qmlpragma Singleton
import Quickshell
import Quickshell.Io
import QtQuick
import qs.config
import qs.services

// The wallpaper list and the current wallpaper. Setting one also regenerates the colour
// scheme with matugen, which rewrites scheme.json; Colours reloads it automatically.
Singleton {
    id: root

    property list<string> list: []
    property alias current: saved.path

    function set(path: string): void {
        saved.path = path;
        Quickshell.execDetached(["matugen", "image", path, "-m", Config.appearance.mode]);
    }

    // regenerate colours for the current wallpaper (e.g. after switching light/dark)
    function regenerate(): void {
        if (current) set(current);
    }

    function rescan(): void {
        scan.running = true;
    }

    Component.onCompleted: rescan()

    Process {
        id: scan
        command: ["sh", "-c", 'find "$1" -maxdepth 2 -type f \\( -iname "*.jpg" -o -iname "*.jpeg" -o -iname "*.png" -o -iname "*.webp" \\) | sort', "sh", Config.paths.wallpaperDir]
        stdout: StdioCollector {
            onStreamFinished: root.list = text.trim().split("\n").filter(f => f)
        }
    }

    FileView {
        path: Colours.stateDir + "/wallpaper.json"
        printErrors: false
        onAdapterUpdated: writeAdapter()
        onLoadFailed: writeAdapter()

        JsonAdapter {
            id: saved
            property string path: ""
        }
    }

    IpcHandler {
        target: "wallpaper"
        function set(path: string): void { root.set(path) }
        function get(): string { return root.current }
    }
}

        }

        component Wall: Image {
            anchors.fill: parent
            fillMode: Image.PreserveAspectCrop
            asynchronous: true
            sourceSize: Qt.size(win.width * win.modelData.devicePixelRatio, win.height * win.modelData.devicePixelRatio)
            Behavior on opacity { Anim { type: Anim.Effects; duration: 800 } }
        }
        readonly property string target: Wallpapers.current ? "file://" + Wallpapers.current : ""

        Wall { id: b; opacity: win.showA ? 0 : 1; onStatusChanged: if (status === Image.Ready && source.toString() === win.target) win.showA = false }

Module 6: Shortcuts and IPC#

cae/daemons/Shortcuts.qmlimport Quickshell
import Quickshell.Io
import Quickshell.Hyprland
import QtQuick
import qs.services
import qs.modules.settings

// Global shortcuts (bind them in Hyprland as `global, cae:<name>`) and IPC for the drawers.
// Both act on the screen that has focus.
Scope {
    id: root

    function toggle(drawer: string): void {
        const s = Visibilities.forFocused();
        if (s && drawer in s) s[drawer] = !s[drawer];
    }

    GShortcut { name: "dashboard"; onPressed: root.toggle("dashboard") }
    GShortcut { name: "launcher"; onPressed: root.toggle("launcher") }
    GShortcut { name: "session"; onPressed: root.toggle("session") }
    GShortcut { name: "sidebar"; onPressed: root.toggle("sidebar") }
    GShortcut { name: "utilities"; onPressed: root.toggle("utilities") }
    GShortcut { name: "settings"; onPressed: Settings.open() }
    GShortcut { name: "showall"; onPressed: { root.toggle("dashboard"); root.toggle("utilities"); root.toggle("osd") } }

    IpcHandler {
        target: "drawers"
        function toggle(drawer: string): void { root.toggle(drawer) }
        function list(): string { return "dashboard launcher session sidebar osd utilities" }
    }

    IpcHandler {
        target: "settings"
        function open(): void { Settings.open() }
    }

    component GShortcut: GlobalShortcut {
        appid: "cae"
    }
}

}

Add a GShortcut { name: "randomWallpaper" } that picks a random file from Wallpapers.list.

✅ Solution
GShortcut {
    name: "randomWallpaper"
    onPressed: Wallpapers.set(Wallpapers.list[Math.floor(Math.random() * Wallpapers.list.length)])
}
