g5Gallery: Desktop Widgets

Widgets that sit on your wallpaper, below your windows: clocks, weather, system rings, graphs, a visualizer, a pomodoro timer. They all use the same recipe: a PanelWindow on the Bottom layer, ExclusionMode.Ignore, a transparent background, and usually mask: Region {} so clicks pass through to the desktop.

A reusable base

Project: Desktop Widgets and Screen Corners builds a DesktopWidget.qml base with Super+drag to move and positions saved to disk. Every widget here can use it instead of a raw PanelWindow.

Big Desktop Clock#

What it is. A huge clock and date on the desktop, with a soft shadow so it's readable on any wallpaper. Seconds are shown small and fade with each tick.

Kind: desktop widget · Needs: nothing extra

galleryshell/desktop/DesktopClock.qmlimport Quickshell
import Quickshell.Wayland
import QtQuick
import QtQuick.Effects

PanelWindow {
    anchors { top: true; left: true }
    margins { top: 140; left: 90 }
    implicitWidth: 520; implicitHeight: 200
    exclusionMode: ExclusionMode.Ignore
    color: "transparent"
    mask: Region {}                                  // click-through
    WlrLayershell.layer: WlrLayer.Bottom             // above wallpaper, below windows
    WlrLayershell.namespace: "quickshell:desktop-clock"

    SystemClock { id: clock; precision: SystemClock.Seconds }

    Item {
        id: content
        anchors.fill: parent
        Row {
            id: timeRow
            spacing: 8
            Text {
                text: Qt.formatDateTime(clock.date, "hh:mm")
                color: "#cba6f7"
                font { family: "Inter"; pixelSize: 110; weight: Font.Black; letterSpacing: -4 }
                font.features: ({ "tnum": 1 })
            }
            Text {
                anchors.baseline: parent.children[0].baseline
                text: String(clock.seconds).padStart(2, "0")
                color: "#cba6f7"; opacity: 0.55
                font { pixelSize: 28; bold: true }
                // fade on each tick
                onTextChanged: tick.restart()
                NumberAnimation on opacity { id: tick; from: 0.9; to: 0.4; duration: 900; running: false }
            }
        }
        Text {
            anchors { top: timeRow.bottom; left: parent.left; leftMargin: 6; topMargin: -12 }
            text: Qt.formatDateTime(clock.date, "dddd, d MMMM")
            color: "#cdd6f4"; font.pixelSize: 24
        }
        layer.enabled: true
        layer.effect: MultiEffect { shadowEnabled: true; shadowBlur: 0.8; shadowOpacity: 0.55; shadowVerticalOffset: 3 }
    }
}

Analog Clock#

What it is. A classic clock face with hour, minute and a smoothly sweeping second hand.

Kind: desktop widget · Needs: nothing extra

galleryshell/desktop/AnalogClock.qmlimport Quickshell
import Quickshell.Wayland
import QtQuick

PanelWindow {
    anchors { top: true; right: true }
    margins { top: 120; right: 120 }
    implicitWidth: 220; implicitHeight: 220
    exclusionMode: ExclusionMode.Ignore
    color: "transparent"
    mask: Region {}
    WlrLayershell.layer: WlrLayer.Bottom

    SystemClock { id: clock; precision: SystemClock.Seconds }

    component Hand: Rectangle {                          // rotated by a `transform` at each use
        property real len
        x: face.width / 2 - width / 2
        y: face.height / 2 - len
        height: len + 10                                  // a bit of tail past the center
        radius: width / 2
    }

    Rectangle {
        id: face
        anchors.fill: parent
        anchors.margins: 10
        radius: width / 2
        color: "#1e1e2e"; border.color: "#45475a"; border.width: 2

        Repeater {                                        // hour ticks
            model: 12
            Rectangle {
                required property int index
                x: face.width / 2 - width / 2; y: 8
                width: index % 3 === 0 ? 4 : 2; height: index % 3 === 0 ? 12 : 6; radius: 2
                color: "#585b70"
                transform: Rotation { origin.x: width / 2; origin.y: face.height / 2 - 8; angle: index * 30 }
            }
        }
        Hand { width: 7; len: face.width * 0.26; color: "#cdd6f4"
               transform: Rotation { origin.x: 3.5; origin.y: face.width * 0.26; angle: (clock.hours % 12 + clock.minutes / 60) * 30 } }
        Hand { width: 5; len: face.width * 0.38; color: "#cdd6f4"
               transform: Rotation { origin.x: 2.5; origin.y: face.width * 0.38; angle: (clock.minutes + clock.seconds / 60) * 6 } }
        Hand {
            id: sec
            width: 2; len: face.width * 0.42; color: "#f38ba8"
            property real a: clock.seconds * 6
            Behavior on a { NumberAnimation { duration: 300; easing.type: Easing.OutBack } }   // "tick" with a little bounce
            transform: Rotation { origin.x: 1; origin.y: face.width * 0.42; angle: sec.a }
        }
        Rectangle { anchors.centerIn: parent; width: 10; height: 10; radius: 5; color: "#f38ba8" }
    }
}

Weather Card#

What it is. A translucent card with the current weather, location, and a three-day forecast.

Kind: desktop widget · Needs: curl (uses wttr.in, no key)

galleryshell/desktop/WeatherCard.qmlimport Quickshell
import Quickshell.Io
import Quickshell.Wayland
import QtQuick
import QtQuick.Layouts

PanelWindow {
    id: root
    property string city: ""
    property var wx: null

    anchors { bottom: true; left: true }
    margins { bottom: 80; left: 80 }
    implicitWidth: 300; implicitHeight: 170
    exclusionMode: ExclusionMode.Ignore
    color: "transparent"
    WlrLayershell.layer: WlrLayer.Bottom

    function icon(code: int): string {
        if (code === 113) return "sunny"
        if (code === 116) return "partly_cloudy_day"
        if (code === 119 || code === 122) return "cloud"
        if ([143, 248, 260].includes(code)) return "foggy"
        if (code >= 386) return "thunderstorm"
        if ([179, 227, 230, 323, 326, 329, 332, 335, 338, 368, 371].includes(code)) return "weather_snowy"
        return "rainy"
    }

    Process {
        id: fetch
        command: ["curl", "-sf", "--max-time", "10", "https://wttr.in/" + encodeURIComponent(root.city) + "?format=j1"]
        stdout: StdioCollector { onStreamFinished: { try { root.wx = JSON.parse(text) } catch (e) {} } }
    }
    Timer { interval: 30 * 60 * 1000; running: true; repeat: true; triggeredOnStart: true; onTriggered: fetch.running = true }

    readonly property var now: wx?.current_condition?.[0] ?? null
    readonly property string place: wx?.nearest_area?.[0]?.areaName?.[0]?.value ?? root.city

    Rectangle {
        anchors.fill: parent
        radius: 24; color: "#cc1e1e2e"; border.color: "#33ffffff"
        ColumnLayout {
            anchors { fill: parent; margins: 18 }
            spacing: 6
            RowLayout {
                spacing: 14
                Text { text: root.icon(parseInt(root.now?.weatherCode ?? 113)); font.family: "Material Symbols Rounded"
                       font.pixelSize: 56; font.variableAxes: ({ "FILL": 1 }); color: "#f9e2af" }
                ColumnLayout {
                    spacing: 0
                    Text { text: (root.now?.temp_C ?? "--") + "°"; color: "#cdd6f4"; font.pixelSize: 40; font.weight: Font.Black }
                    Text { text: (root.now?.weatherDesc?.[0]?.value ?? "") + " · " + root.place; color: "#a6adc8"; font.pixelSize: 12 }
                }
            }
            RowLayout {
                spacing: 18
                Repeater {
                    model: root.wx?.weather ?? []
                    Text {
                        required property var modelData
                        text: Qt.formatDate(new Date(modelData.date), "ddd") + "  " + modelData.maxtempC + "° / " + modelData.mintempC + "°"
                        color: "#a6adc8"; font.pixelSize: 11
                    }
                }
            }
        }
    }
}

cava is a terminal audio visualizer that can print raw numbers. The shell script writes a small config to the runtime directory (raw ASCII output, values 0–100, separated by ; which is character 59), then execs cava with it.
