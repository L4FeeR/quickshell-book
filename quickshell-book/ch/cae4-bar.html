cae4Caelestia 4: The Vertical Bar and Its Popouts

Caelestia's bar runs down the left side of the screen, inside the frame. It's narrow (40 px of content), so every module is designed for a column: workspaces stack vertically, the window title is written sideways, and the clock shows hours above minutes. Hovering a status icon slides a popout out of the bar with details.

modules/bar/      BarWrapper  Bar  OsIcon  Workspaces  ActiveWindow  Tray  Clock  StatusIcons  Power
modules/popouts/  Wrapper  AudioPopout  NetworkPopout  BluetoothPopout  BatteryPopout

Module 1: The bar's slot#

The bar lives in the frame's left side. When bar.persistent is false, it shrinks to the frame thickness and slides out when you touch the left edge.

cae/modules/bar/BarWrapper.qmlimport Quickshell
import QtQuick
import qs.config
import qs.components

// The bar's slot in the frame. Its width animates between the frame thickness (hidden)
// and the full bar width (shown), and the real bar is only loaded while it's visible.
Item {
    id: root

    required property ShellScreen screen
    required property ScreenState screenState
    required property Item popouts

    property bool hovered: false
    readonly property int padding: Math.max(Tokens.padding.small, Config.border.thickness)
    readonly property int contentWidth: Tokens.sizes.barInnerWidth + padding * 2
    readonly property bool shown: Config.bar.persistent || screenState.bar || hovered
    readonly property int exclusiveZone: Config.bar.persistent || screenState.bar ? contentWidth : Config.border.thickness

    function checkPopout(y: real): void {
        loader.item?.checkPopout(y);
    }

    implicitWidth: shown ? contentWidth : Config.border.thickness
    clip: true

    Behavior on implicitWidth {
        Anim { type: root.shown ? Anim.Spatial : Anim.Emphasized }
    }

    Loader {
        id: loader

        anchors.top: parent.top
        anchors.bottom: parent.bottom
        anchors.right: parent.right
        active: root.shown || root.implicitWidth > Config.border.thickness

        sourceComponent: Bar {
            width: root.contentWidth
            screen: root.screen
            screenState: root.screenState
            popouts: root.popouts
        }
    }
}

Module 2: The bar, from a list of entries#

cae/modules/bar/Bar.qmlimport Quickshell
import QtQuick
import QtQuick.Layouts
import qs.config
import qs.components

// The vertical bar. Which modules appear, and in what order, comes from Config.bar.entries,
// e.g. ["logo", "workspaces", "spacer", "activeWindow", "spacer", "tray", "clock", "statusIcons", "power"].
ColumnLayout {
    id: root

    required property ShellScreen screen
    required property ScreenState screenState
    required property Item popouts

    // Called while the pointer moves over the bar: open the popout of the status icon under it.
    function checkPopout(y: real): void {
        const entry = childAt(width / 2, y);
        const icons = entry?.item;
        if (entry?.entryId === "statusIcons" && icons) {
            const hit = icons.iconAt(icons.mapFromItem(root, 0, y).y);
            if (hit) {
                popouts.currentName = hit.name;
                popouts.currentCenter = hit.mapToItem(root, 0, hit.height / 2).y;
                popouts.hasCurrent = true;
                return;
            }
        }
        popouts.hasCurrent = false;
    }

    spacing: Tokens.spacing.medium

    Repeater {
        model: Config.bar.entries

        Loader {
            required property string modelData
            readonly property string entryId: modelData

            Layout.alignment: Qt.AlignHCenter
            Layout.fillHeight: modelData === "spacer"
            Layout.topMargin: index === 0 ? Tokens.padding.large : 0
            Layout.bottomMargin: index === Config.bar.entries.length - 1 ? Tokens.padding.large : 0

            required property int index

            sourceComponent: ({
                spacer: spacer, logo: logo, workspaces: workspaces, activeWindow: activeWindow,
                tray: tray, clock: clock, statusIcons: statusIcons, power: power
            })[modelData] ?? null
        }
    }

    Component { id: spacer; Item {} }
    Component { id: logo; OsIcon { screenState: root.screenState } }
    Component { id: workspaces; Workspaces { screen: root.screen } }
    Component { id: activeWindow; ActiveWindow { bar: root } }
    Component { id: tray; Tray {} }
    Component { id: clock; Clock {} }
    Component { id: statusIcons; StatusIcons {} }
    Component { id: power; Power { screenState: root.screenState } }
}

Module 3: The logo#

cae/modules/bar/OsIcon.qmlimport Quickshell
import Quickshell.Io
import Quickshell.Widgets
import QtQuick
import qs.components

// Your distro's logo from the icon theme. Click it to open the launcher.
Item {
    id: root

    required property ScreenState screenState
    property string distro: "linux"

    implicitWidth: 26
    implicitHeight: 26

    IconImage {
        anchors.fill: parent
        source: Quickshell.iconPath(`distributor-logo-${root.distro}`, Quickshell.iconPath("distributor-logo", "start-here"))
    }

    StateLayer {
        anchors.margins: -4
        radius: width / 2
        onClicked: root.screenState.launcher = !root.screenState.launcher
    }

    FileView {
        path: "/etc/os-release"
        onLoaded: root.distro = (text().match(/^ID=(.*)$/m)?.[1] ?? "linux").replace(/"/g, "")
    }
}

Your distro's ID comes from /etc/os-release (ID=arch), and icon themes usually ship a distributor-logo-arch icon. If not, it falls back to the generic distributor-logo, then start-here. The StateLayer gets negative margins so the round hover highlight is bigger than the icon.

Module 4: Workspaces#

The most detailed module of the bar. It shows a group of workspaces (5 by default: 1–5, then 6–10 when you move to 6), with three layers:

cae/modules/bar/Workspaces.qmlimport Quickshell
import Quickshell.Hyprland
import Quickshell.Widgets
import QtQuick
import qs.config
import qs.services
import qs.components

// Caelestia-style vertical workspaces: a group of `shown` workspaces, merged backgrounds behind
// occupied ones, a pill on the active one that stretches as it moves ("trail"), and app icons.
StyledRect {
    id: root

    required property ShellScreen screen

    readonly property HyprlandMonitor monitor: Hyprland.monitorFor(screen)
    readonly property int shown: Config.bar.workspacesShown
    readonly property int activeId: monitor?.activeWorkspace?.id ?? 1
    readonly property int groupStart: Math.floor((activeId - 1) / shown) * shown + 1
    readonly property int slot: Tokens.sizes.barInnerWidth - Tokens.padding.small * 2

    function workspace(id: int): HyprlandWorkspace {
        return Hyprland.workspaces.values.find(w => w.id === id) ?? null;
    }
    function occupied(id: int): bool {
        return (workspace(id)?.toplevels.values.length ?? 0) > 0;
    }

    implicitWidth: Tokens.sizes.barInnerWidth
    implicitHeight: column.implicitHeight + Tokens.padding.small * 2
    radius: Tokens.rounding.full
    color: Colours.palette.m3surfaceContainer

    // merged backgrounds: each occupied slot stretches toward occupied neighbours
    Repeater {
        model: root.shown

        StyledRect {
            required property int index
            readonly property int wsId: root.groupStart + index
            readonly property bool occ: root.occupied(wsId)
            readonly property bool prevOcc: index > 0 && root.occupied(wsId - 1)
            readonly property bool nextOcc: index < root.shown - 1 && root.occupied(wsId + 1)
            readonly property Item cell: cells.count > index ? cells.itemAt(index) : null   // count makes the binding update

            x: Tokens.padding.small
            width: root.slot
            y: (cell?.y ?? 0) + Tokens.padding.small - (prevOcc ? column.spacing / 2 + 1 : 0)
            height: occ ? (cell?.height ?? 0) + (prevOcc ? column.spacing / 2 + 1 : 0) + (nextOcc ? column.spacing / 2 + 1 : 0) : 0
            radius: Tokens.rounding.full
            topLeftRadius: prevOcc ? 0 : radius
            topRightRadius: prevOcc ? 0 : radius
            bottomLeftRadius: nextOcc ? 0 : radius
            bottomRightRadius: nextOcc ? 0 : radius
            color: Colours.palette.m3surfaceContainerHighest
            opacity: occ ? 1 : 0

            Behavior on opacity { Anim { type: Anim.Effects } }
        }
    }

    // the active pill with a trail: the leading edge moves fast, the trailing edge catches up
    StyledRect {
        id: active

        readonly property Item cell: cells.count > 0 ? cells.itemAt(root.activeId - root.groupStart) : null
        readonly property real targetTop: (cell?.y ?? 0) + Tokens.padding.small
        readonly property real targetBottom: targetTop + (cell?.height ?? root.slot)
        property real topEdge: targetTop
        property real bottomEdge: targetBottom
        property bool movingDown: false

        onTargetTopChanged: movingDown = targetTop > topEdge

        x: Tokens.padding.small
        width: root.slot
        y: topEdge
        height: Math.max(width, bottomEdge - topEdge)
        radius: Tokens.rounding.full
        color: Colours.palette.m3primary

        Behavior on topEdge {
            Anim { type: Anim.Spatial; duration: active.movingDown ? Tokens.anim.durations.slowSpatial : Tokens.anim.durations.fastSpatial }
        }
        Behavior on bottomEdge {
            Anim { type: Anim.Spatial; duration: active.movingDown ? Tokens.anim.durations.fastSpatial : Tokens.anim.durations.slowSpatial }
        }
    }

    Column {
        id: column

        anchors.horizontalCenter: parent.horizontalCenter
        y: Tokens.padding.small
        spacing: Tokens.spacing.small

        Repeater {
            id: cells
            model: root.shown

            Item {
                id: cell

                required property int index
                readonly property int wsId: root.groupStart + index
                readonly property bool isActive: wsId === root.activeId
                readonly property var windows: root.workspace(wsId)?.toplevels.values ?? []

                implicitWidth: root.slot
                implicitHeight: root.slot + (Config.bar.workspaceWindows ? icons.implicitHeight : 0)

                StyledText {
                    id: label
                    anchors.horizontalCenter: parent.horizontalCenter
                    y: (root.slot - height) / 2
                    text: cell.wsId
                    font.pixelSize: Tokens.font.size.small
                    font.bold: cell.isActive
                    color: cell.isActive ? Colours.palette.m3onPrimary
                         : root.occupied(cell.wsId) ? Colours.palette.m3onSurface : Colours.palette.m3outline
                }

                Column {
                    id: icons
                    anchors.horizontalCenter: parent.horizontalCenter
                    y: root.slot - Tokens.padding.extraSmall
                    spacing: 2
                    bottomPadding: Tokens.padding.small
                    visible: Config.bar.workspaceWindows

                    Repeater {
                        model: cell.windows.slice(0, 3)

                        IconImage {
                            required property HyprlandToplevel modelData
                            readonly property string appId: modelData.wayland?.appId ?? modelData.lastIpcObject?.class ?? ""
                            implicitSize: 14
                            source: Quickshell.iconPath(DesktopEntries.heuristicLookup(appId)?.icon ?? appId, "application-x-executable")
                        }
                    }
                }

                StateLayer {
                    radius: Tokens.rounding.full
                    onClicked: Hypr.workspace(cell.wsId)
                }
            }
        }
    }

    WheelHandler {
        onWheel: event => Hypr.workspaceRelative(event.angleDelta.y > 0 ? -1 : 1)
    }
}
