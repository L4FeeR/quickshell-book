g7Gallery: More Bar Modules

Fifteen more chips for your bar: network, Bluetooth, brightness, microphone, a scrolling media title, a taskbar, a tray with an overflow drawer, workspaces with app icons, and small tools like a stopwatch and a countdown. Each is one file in galleryshell/bar/. Drop it into a bar RowLayout, or see all of them working together in Gallery: Build a Whole Shell from the Gallery.

Icons

Like the rest of the gallery, these chips draw icons with the Material Symbols Rounded font (Fonts and Icons). Without it installed you'll see the icon names as words (volume_up), which is a quick way to tell the font is missing.

Network Chip#

What it is. Shows how you're online: Wi-Fi strength and network name, "Ethernet" when wired, or a crossed-out icon when offline. Click hides the name, right-click turns Wi-Fi off and on.

Kind: bar module · Needs: NetworkManager

import QtQuick

Brightness Chip#

What it is. Screen brightness for laptops. Scroll over it to change brightness in 5% steps; click to jump between dim and full.

Kind: bar module · Needs: brightnessctl, a backlight (laptops; most external monitors need ddcutil instead)

galleryshell/bar/BrightnessChip.qmlimport Quickshell
import Quickshell.Io
import QtQuick
import QtQuick.Layouts

Rectangle {
    id: root
    property real level: 0                          // 0..1
    property string device: ""                      // "" = brightnessctl's default backlight

    function args(extra: list<string>): list<string> {
        return device ? ["brightnessctl", "-d", device].concat(extra) : ["brightnessctl"].concat(extra)
    }
    function set(v: real): void {
        level = Math.max(0.01, Math.min(1, v))
        Quickshell.execDetached(args(["-q", "set", Math.round(level * 100) + "%"]))
    }

    // brightnessctl -m prints: name,class,current,percent,max  e.g. "intel_backlight,backlight,400,42%,960"
    Process {
        id: read
        command: root.args(["-m"])
        running: true
        stdout: StdioCollector {
            onStreamFinished: {
                const f = text.trim().split(",")
                if (f.length >= 5) root.level = Number(f[2]) / Number(f[4])
            }
        }
    }
    // other tools (keys, other widgets) may change it too: re-read now and then
    Timer { interval: 5000; running: true; repeat: true; onTriggered: read.running = true }

    implicitWidth: row.implicitWidth + 20
    implicitHeight: 26
    radius: height / 2
    color: hover.hovered ? "#45475a" : "#313244"

    RowLayout {
        id: row
        anchors.centerIn: parent
        spacing: 6
        Text {
            text: root.level > 0.66 ? "brightness_high" : root.level > 0.33 ? "brightness_medium" : "brightness_low"
            font.family: "Material Symbols Rounded"; font.pixelSize: 18
            color: "#f9e2af"
        }
        Text {
            text: Math.round(root.level * 100) + "%"
            color: "#cdd6f4"; font.pixelSize: 12
            font.features: ({ "tnum": 1 })
        }
    }

    HoverHandler { id: hover }
    WheelHandler { onWheel: e => root.set(root.level + e.angleDelta.y / 120 * 0.05) }
    TapHandler { onTapped: root.set(root.level > 0.5 ? 0.2 : 1) }
}

What it is. A stopwatch chip. Click to start and pause, right-click to reset. It turns green while running.

Kind: bar module · Needs: nothing extra

galleryshell/bar/Stopwatch.qmlimport QtQuick
import QtQuick.Layouts

// Click: start/pause. Right-click: reset. Shows mm:ss (and h: once past an hour).
Rectangle {
    id: root
    property bool running: false
    property real elapsed: 0          // ms counted before the current run
    property real startedAt: 0        // Date.now() when the current run started
    property real now: Date.now()

    readonly property real total: elapsed + (running ? now - startedAt : 0)

    function fmt(ms: real): string {
        const s = Math.floor(ms / 1000), h = Math.floor(s / 3600), m = Math.floor(s / 60) % 60
        const pad = n => String(n).padStart(2, "0")
        return (h > 0 ? h + ":" : "") + pad(m) + ":" + pad(s % 60)
    }
    function toggle(): void {
        if (running) { elapsed += Date.now() - startedAt; running = false }
        else { startedAt = Date.now(); now = startedAt; running = true }
    }

    // only ticks while running: a stopped stopwatch costs nothing
    Timer { interval: 200; running: root.running; repeat: true; onTriggered: root.now = Date.now() }

    implicitWidth: row.implicitWidth + 20
    implicitHeight: 26
    radius: height / 2
    color: root.running ? "#a6e3a1" : hover.hovered ? "#45475a" : "#313244"
    Behavior on color { ColorAnimation { duration: 150 } }

        id: row
        anchors.centerIn: parent
