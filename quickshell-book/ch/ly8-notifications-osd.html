ly8lyne-dots 8: Notifications, Audio and the OSD

Two things every desktop needs: somewhere for notifications to go, and feedback when you press the volume keys. This chapter turns the shell into the system's notification daemon (replacing mako or dunst), with popups, a history under the bell and inline replies, then builds the audio and brightness services and lyne's OSD, which slides out of the bar like the other panels.

The notification service#

lyne/services/NotificationService.qmlpragma Singleton
import QtQuick
import Quickshell
import qs.services
import Quickshell.Io
import Quickshell.Services.Notifications
import qs.config

// The notification daemon: receives every notification, keeps the history,
// and decides which ones pop up. Only ONE notification server may run in a
// session, which is why it lives in a singleton.
Singleton {
    id: root

    readonly property bool dnd: StateService.get("notifications.dnd", false)
    readonly property var history: server.trackedNotifications.values    // oldest first
    readonly property int count: history.length
    property var popups: []                       // the ones showing as popups right now

    // the history popup under the bell
    property bool historyVisible: false
    property string historyScreen: ""

    NotificationServer {
        id: server
        keepOnReload: true
        actionsSupported: true
        imageSupported: true
        bodyMarkupSupported: true
        inlineReplySupported: true                // apps like chat clients can offer a reply field
        persistenceSupported: true

        onNotification: n => {
            n.tracked = true;                     // keep it, or it's dropped right away
            root._arrived[n.id] = Date.now();     // notifications carry no time stamp: record it
            n.closed.connect(() => root.hidePopup(n));
            if (root.dnd && n.urgency !== NotificationUrgency.Critical)
                return;                           // do not disturb: history only
            const list = root.popups.concat([n]);
            root.popups = list.slice(-Config.notifMaxPopups);    // never more than maxPopups
        }
    }

    function hidePopup(n) { popups = popups.filter(x => x !== n); }
    function clearAll() { history.slice().forEach(n => n.dismiss()); }
    function toggleDnd() { StateService.set("notifications.dnd", !dnd); }

    function toggleHistory(screen) {
        if (historyVisible && historyScreen === screen) { historyVisible = false; return; }
        WindowManagerService.openExclusive("Notifications", () => { historyScreen = screen; historyVisible = true; });
    }

    // how long ago a notification arrived: "now", "5 min", "2 h"
    property var _arrived: ({})
    function age(n) {
        const s = Math.floor((Date.now() - (_arrived[n.id] ?? Date.now())) / 1000);
        return s < 60 ? "now" : s < 3600 ? Math.floor(s / 60) + " min" : Math.floor(s / 3600) + " h";
    }

    Connections {
        target: WindowManagerService
        function onCloseRequested(except) { if (except !== "Notifications") root.historyVisible = false; }
    }
    onHistoryVisibleChanged: historyVisible ? WindowManagerService.registerOpen("Notifications") : WindowManagerService.registerClose("Notifications")

    IpcHandler {
        target: "notifications"
        function toggleDnd(): void { root.toggleDnd(); }
        function clear(): void { root.clearAll(); }
    }
}
Code walkthrough · piece by piece
lines 9–11One server per session
// The notification daemon: receives every notification, keeps the history,
// and decides which ones pop up. Only ONE notification server may run in a
// session, which is why it lives in a singleton.

The freedesktop notification protocol has one server per session: whoever registers first gets every notification. A singleton guarantees the shell creates exactly one, and every widget (popups, history, bell) reads from it.

lines 15–18State
readonly property bool dnd: StateService.get("notifications.dnd", false)
readonly property var history: server.trackedNotifications.values    // oldest first
readonly property int count: history.length
property var popups: []                       // the ones showing as popups right now

Do not disturb is a setting, so it survives restarts. history is every notification still open, count feeds the bell, popups is the subset showing as popups.

lines 20–22The history popup
// the history popup under the bell
property bool historyVisible: false
property string historyScreen: ""

Open or not, and on which screen.

lines 24–31The server
NotificationServer {
    id: server
    keepOnReload: true
    actionsSupported: true
    imageSupported: true
    bodyMarkupSupported: true
    inlineReplySupported: true                // apps like chat clients can offer a reply field
    persistenceSupported: true

keepOnReload keeps notifications when the config reloads. The …Supported flags tell apps what the shell can show; apps then send actions, images, simple markup, and a reply field (chat apps use it).

lines 33–36A new notification
onNotification: n => {
    n.tracked = true;                     // keep it, or it's dropped right away
    root._arrived[n.id] = Date.now();     // notifications carry no time stamp: record it
    n.closed.connect(() => root.hidePopup(n));

tracked = true keeps it; without it the notification is dropped as soon as this handler returns. Notifications carry no time stamp, so the arrival time is recorded. When the notification closes (dismissed, expired or withdrawn by the app), it leaves the popups too.

lines 37–38Do not disturb
if (root.dnd && n.urgency !== NotificationUrgency.Critical)
    return;                           // do not disturb: history only

Only critical notifications pop up; the rest go quietly into the history.

lines 39–40Never too many
const list = root.popups.concat([n]);
root.popups = list.slice(-Config.notifMaxPopups);    // never more than maxPopups

New popups are added at the end, and only the last maxPopups are kept.

lines 44–46Helpers
function hidePopup(n) { popups = popups.filter(x => x !== n); }
function clearAll() { history.slice().forEach(n => n.dismiss()); }
function toggleDnd() { StateService.set("notifications.dnd", !dnd); }

Hide a popup (it stays in the history), dismiss everything (slice() copies the list first, because dismissing changes it), and flip do not disturb.

The same pattern as every popup.

Only one daemon

Stop mako, dunst or swaync before running the shell, or the shell logs that the server is already registered and stays quiet until the other one exits.

The popups#

lyne/modules/notifications/NotificationOverlay.qmlimport QtQuick
import Quickshell
import Quickshell.Wayland
import Quickshell.Hyprland
import qs.config
import qs.services

// The popup stack at the top right of the focused monitor. Each card slides
// in from the right, lives for the notification's timeout (hovering holds
// it), then leaves; it stays in the history either way.
PanelWindow {
    id: root

    screen: Quickshell.screens.find(s => s.name === Hyprland.focusedMonitor?.name) ?? Quickshell.screens[0]
    anchors {
        top: !Config.barOnBottom
        bottom: Config.barOnBottom
        right: true
    }
    margins.top: Config.barOnBottom ? 0 : Config.barReservedHeight + Config.spacing
    margins.bottom: Config.barOnBottom ? Config.barReservedHeight + Config.spacing : 0
    margins.right: Config.spacing
    exclusionMode: ExclusionMode.Ignore
    implicitWidth: Config.notifWidth
    implicitHeight: Math.max(1, stack.implicitHeight)
    visible: NotificationService.popups.length > 0
    color: "transparent"
    WlrLayershell.layer: WlrLayer.Overlay
    WlrLayershell.namespace: "lyne_notifications"
    // the keyboard only while typing a reply, so popups never steal focus
    WlrLayershell.keyboardFocus: WlrKeyboardFocus.OnDemand

    Column {
        id: stack
        width: parent.width
        spacing: Config.spacing

        Repeater {
            model: NotificationService.popups
            NotificationCard {
                id: card
                required property var modelData
                notif: modelData
                width: stack.width
                color: Config.backgroundTransparentColor

                // slide in from the right
                transform: Translate { id: shift; x: root.width }
                Component.onCompleted: slideIn.start()
                NumberAnimation { id: slideIn; target: shift; property: "x"; to: 0; duration: Config.animDurationLong; easing.type: Easing.OutExpo }

                readonly property int timeout: card.critical ? 0 : (modelData.expireTimeout > 0 ? modelData.expireTimeout : Config.notifTimeout)
                Timer {
                    interval: card.timeout
                    running: card.timeout > 0 && !hover.hovered
                    onTriggered: NotificationService.hidePopup(card.modelData)
                }
                HoverHandler { id: hover }

// Laptop screen brightness through brightnessctl.
Singleton {
    id: root

    property real brightness: 0           // 0..1
    property bool available: false        // false on desktops without a backlight
    readonly property real step: 0.05

    function setBrightness(v) {
        brightness = Math.max(0.01, Math.min(1, v));
        Quickshell.execDetached(["brightnessctl", "-q", "set", Math.round(brightness * 100) + "%"]);
    }
    function increaseBrightness() { setBrightness(brightness + step); }
    function decreaseBrightness() { setBrightness(brightness - step); }

    // `brightnessctl -m` prints "name,class,current,percent,max"
    Process {
        id: reader
        running: true
        command: ["brightnessctl", "-m", "-c", "backlight"]
        stdout: StdioCollector {
            onStreamFinished: {
                const f = text.trim().split(",");
                root.available = f.length >= 5;
                if (root.available)
                    root.brightness = Number(f[2]) / Number(f[4]);
            }
        }
    }
    Timer { interval: 10000; running: root.available; repeat: true; onTriggered: reader.running = true }
}

Brightness through brightnessctl: -m gives machine-readable output (name,class,current,percent,max), read at startup and every 10 seconds; available is false on desktops without a backlight, which hides the brightness slider. // The on-screen display for volume, microphone and brightness.

Singleton {
    id: root

    property bool visible: false          // what the OSD wants
    property bool mapped: false           // the window exists (stays a bit longer for the exit animation)

    readonly property real value: kind === "brightness" ? BrightnessService.brightness
        : kind === "mic" ? (AudioService.micMuted ? 0 : AudioService.micVolume)
        : (AudioService.muted ? 0 : AudioService.volume)
    readonly property string icon: kind === "brightness" ? "\u{f00df}"
        : kind === "mic" ? (AudioService.micMuted ? "\u{f036d}" : "\u{f036c}")
        : AudioService.volumeIcon(AudioService.volume, AudioService.muted)
    readonly property string label: kind === "brightness" ? "Brightness" : kind === "mic" ? "Microphone" : "Volume"

    function show(k) {
        kind = k;
        mapped = true;
        visible = true;
        hideTimer.restart();
    }
    function keepOpen() { hideTimer.restart(); }

    Timer {
        id: hideTimer
        interval: Config.osdTimeout
        onTriggered: { root.visible = false; unmapTimer.restart(); }
    }
    // unmap only after the exit animation has played
    Timer {
        id: unmapTimer
        interval: Config.animDurationLong
        onTriggered: if (!root.visible) root.mapped = false
    }

    // scripts can show it too: qs -c lyne ipc call osd trigger volume
    IpcHandler {
        target: "osd"
        function trigger(kind: string): void { root.show(kind); }
    }
}
