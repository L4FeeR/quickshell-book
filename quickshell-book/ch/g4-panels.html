g4Gallery: Panels and Menus

Bigger pieces of UI that open on demand: quick settings, an app grid, a power menu, pickers. Each is shown standalone with an IPC toggle so you can open it with qs -c gallery ipc call <name> toggle. For the full versions wired into a complete shell, see Part VI.

Quick Settings Tiles#

What it is. A grid of toggle tiles (Wi-Fi, Bluetooth, DND, night light, caffeine, mic) plus volume and brightness sliders. Active tiles fill with the accent color and morph from pill to rounded square.

Kind: its own window · Needs: NetworkManager, BlueZ, PipeWire (each tile hides itself if its service is missing)

galleryshell/panels/QuickSettings.qmlimport Quickshell
import Quickshell.Io
import Quickshell.Wayland
import Quickshell.Bluetooth
import Quickshell.Networking
import Quickshell.Services.Pipewire
import QtQuick
import QtQuick.Layouts
import QtQuick.Controls

Scope {
    id: root
    property bool open: false
    property bool dnd: false
    property bool caffeine: false
    readonly property PwNode sink: Pipewire.defaultAudioSink
    readonly property PwNode mic: Pipewire.defaultAudioSource
    PwObjectTracker { objects: [root.sink, root.mic] }

    IpcHandler { target: "quicksettings"; function toggle(): void { root.open = !root.open } }

    component Tile: Rectangle {
        id: tile
        property string icon
        property string label
        property bool active
        signal toggled()
        Layout.fillWidth: true
        implicitHeight: 56
        radius: active ? 16 : height / 2
        color: active ? "#cba6f7" : "#313244"
        Behavior on radius { NumberAnimation { duration: 250; easing.type: Easing.OutCubic } }
        Behavior on color { ColorAnimation { duration: 200 } }
        RowLayout {
            anchors { fill: parent; leftMargin: 16; rightMargin: 12 }
            spacing: 10
            Text { text: tile.icon; font.family: "Material Symbols Rounded"; font.pixelSize: 20
                   font.variableAxes: ({ "FILL": tile.active ? 1 : 0 }); color: tile.active ? "#1e1e2e" : "#cdd6f4" }
            Text { Layout.fillWidth: true; text: tile.label; elide: Text.ElideRight; font.pixelSize: 13
                   color: tile.active ? "#1e1e2e" : "#cdd6f4"; font.bold: tile.active }
        }
        TapHandler { onTapped: tile.toggled() }
    }

    component SliderRow: RowLayout {
        id: sr
        property string icon
        property real value
        signal moved(real v)
        Layout.fillWidth: true
        spacing: 10
        Text { text: sr.icon; font.family: "Material Symbols Rounded"; font.pixelSize: 20; color: "#cdd6f4" }
        Slider {
            id: s
            Layout.fillWidth: true
            from: 0; to: 1; value: sr.value
            onMoved: sr.moved(value)
            background: Rectangle {
                y: s.topPadding + s.availableHeight / 2 - height / 2
                width: s.availableWidth; height: 20; radius: 10; color: "#313244"
                Rectangle { width: Math.max(height, s.visualPosition * parent.width); height: parent.height; radius: 10; color: "#cba6f7" }
            }
            handle: Item {}
        }
    }

    PanelWindow {
        id: panel; anchors { top: true; right: true }
        margins { top: 54; right: 10 }
        implicitWidth: 360; implicitHeight: col.implicitHeight + 32
        exclusionMode: ExclusionMode.Ignore
        color: "transparent"
        visible: root.open
        WlrLayershell.layer: WlrLayer.Overlay
        WlrLayershell.namespace: "quickshell:quicksettings"

        Rectangle {
            anchors.fill: parent
            radius: 24; color: "#1e1e2e"; border.color: "#313244"
            ColumnLayout {
                id: col
                anchors { left: parent.left; right: parent.right; top: parent.top; margins: 16 }
                spacing: 10
                GridLayout {
                    Layout.fillWidth: true
                    columns: 2; rowSpacing: 8; columnSpacing: 8
                    Tile {
                        readonly property var net: Networking.devices.values.find(d => d.type === DeviceType.Wifi)?.networks.values.find(n => n.connected)
                        icon: "wifi"; label: net?.name ?? "Wi-Fi"
                        active: Networking.wifiEnabled
                        onToggled: Networking.wifiEnabled = !Networking.wifiEnabled
                    }
                    Tile {
                        visible: Bluetooth.defaultAdapter !== null
                        icon: "bluetooth"; label: Bluetooth.devices.values.find(d => d.connected)?.name ?? "Bluetooth"
                        active: Bluetooth.defaultAdapter?.enabled ?? false
                        onToggled: Bluetooth.defaultAdapter.enabled = !Bluetooth.defaultAdapter.enabled
                    }
                    Tile { icon: "do_not_disturb_on"; label: "Do not disturb"; active: root.dnd; onToggled: root.dnd = !root.dnd }
                    Tile {
                        icon: "nightlight"; label: "Night light"; active: night.running
                        onToggled: night.running = !night.running
                        Process { id: night; command: ["hyprsunset", "-t", "4200"] }
                    }
                    Tile {
                        icon: "coffee"; label: "Caffeine"; active: root.caffeine
                        onToggled: root.caffeine = !root.caffeine
                    }
                    Tile {
                        icon: root.mic?.audio?.muted ? "mic_off" : "mic"; label: "Microphone"
                        active: !(root.mic?.audio?.muted ?? true)
                        onToggled: if (root.mic?.ready) root.mic.audio.muted = !root.mic.audio.muted
                    }
                }
                SliderRow {
                    icon: "volume_up"
                    value: root.sink?.audio?.volume ?? 0
                    onMoved: v => { if (root.sink?.ready) root.sink.audio.volume = v }
                }
            }
            IdleInhibitor { window: panel; enabled: root.caffeine }
        }
    }
}

The IdleInhibitor is attached to this panel's window (panel, the id on line 68) and enabled by the caffeine tile. App Grid Launcher#

What it is. A full-screen-style launcher showing installed apps as a grid of big icons; type to filter, use arrow keys to move, Enter to launch.

Kind: its own window · Needs: nothing extra

galleryshell/panels/AppGrid.qmlimport Quickshell
import Quickshell.Io
import Quickshell.Wayland
import Quickshell.Widgets
import QtQuick
import QtQuick.Layouts

Scope {
    id: root
    property bool open: false
    IpcHandler { target: "appgrid"; function toggle(): void { root.open = !root.open } }

    LazyLoader {
        active: root.open
        PanelWindow {
            id: win
            implicitWidth: 760; implicitHeight: 560
            exclusionMode: ExclusionMode.Ignore
            color: "transparent"
            WlrLayershell.layer: WlrLayer.Overlay
            WlrLayershell.keyboardFocus: WlrKeyboardFocus.Exclusive

            readonly property var apps: DesktopEntries.applications.values
                .filter(a => a.name.toLowerCase().includes(search.text.toLowerCase())
                          || (a.genericName ?? "").toLowerCase().includes(search.text.toLowerCase()))
                .sort((a, b) => a.name.localeCompare(b.name))
            function launch(e) { Quickshell.execDetached(e.command); root.open = false }

            Rectangle {
                anchors.fill: parent
                radius: 28; color: "#ee1e1e2e"; border.color: "#313244"
                ColumnLayout {
                    anchors { fill: parent; margins: 24 }
                    spacing: 16
                    Rectangle {
                        Layout.fillWidth: true; implicitHeight: 46; radius: 23; color: "#313244"
                        TextInput {
                            id: search
                            anchors { fill: parent; leftMargin: 20; rightMargin: 20 }
                            verticalAlignment: TextInput.AlignVCenter
                            color: "#cdd6f4"; font.pixelSize: 16; focus: true
                            Text { visible: !parent.text; text: "Search apps…"; color: "#6c7086"; font: parent.font; anchors.verticalCenter: parent.verticalCenter }
                            onTextChanged: grid.currentIndex = 0
                            Keys.onEscapePressed: root.open = false
                            Keys.onReturnPressed: if (grid.currentIndex >= 0) win.launch(grid.model.values[grid.currentIndex])
                            Keys.forwardTo: [grid]                          // arrows move the grid selection
                        }
                    }
                    GridView {
                        id: grid
                        Layout.fillWidth: true; Layout.fillHeight: true
                        clip: true
                        cellWidth: width / 6; cellHeight: 112
                        keyNavigationEnabled: true
                        model: ScriptModel { values: win.apps }
                        highlight: Rectangle { radius: 18; color: "#313244" }
                        highlightMoveDuration: 100
                        delegate: Item {
                            required property DesktopEntry modelData
                            required property int index
                            width: grid.cellWidth; height: grid.cellHeight
                            Column {
                                anchors.centerIn: parent
                                spacing: 8
                                IconImage {
                                    anchors.horizontalCenter: parent.horizontalCenter
                                    implicitSize: 52
                                    source: Quickshell.iconPath(modelData.icon, "application-x-executable")
                                    asynchronous: true
                                    scale: ma.containsMouse ? 1.12 : 1
                                    Behavior on scale { NumberAnimation { duration: 150; easing.type: Easing.OutBack } }
                                }
                                Text {
                                    width: grid.cellWidth - 12
                                    horizontalAlignment: Text.AlignHCenter
                                    text: modelData.name; color: "#cdd6f4"; font.pixelSize: 12; elide: Text.ElideRight
                                }
                            }
                            MouseArea {
                                id: ma; anchors.fill: parent; hoverEnabled: true
                                onEntered: grid.currentIndex = index
                                onClicked: { Quickshell.execDetached(modelData.command); root.open = false }
                            }
                        }
                    }
                }
            }
        }
    }
}
