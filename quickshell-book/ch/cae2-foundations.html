cae2Caelestia 2: Tokens, Colours and Building Blocks

Before any panel, Caelestia sets up a small design system. Every size, radius, font, colour and animation in the shell comes from three places: Tokens (numbers), Colours (the Material 3 palette) and a handful of building blocks that already use both. Build these first and the rest of the shell becomes short and consistent.

config/      Tokens.qml  Config.qml
components/  Anim  CAnim  StyledRect  StyledText  MaterialIcon  StateLayer  IconButton  ScreenState
services/    Colours  Visibilities  Hypr

Module 1: Tokens#

Material 3 names its sizes on a fixed scale (extra small 4 px, small 8, medium 12, large 16, extra large 28). Caelestia uses exactly these names, and so does the rebuild.

cae/config/Tokens.qmlpragma Singleton
import Quickshell
import QtQuick
import qs.config

// Design tokens: every size, radius, font and motion value in the shell comes from here.
// The names follow Material 3 (and Caelestia's own token names).
Singleton {
    id: root

    // Two user options scale the tokens live (see the settings app).
    readonly property real r: Config.appearance.roundingScale
    readonly property real speed: Math.max(0.1, Config.appearance.animationSpeed)

    readonly property QtObject rounding: QtObject {
        readonly property int extraSmall: Math.round(4 * root.r)
        readonly property int small: Math.round(8 * root.r)
        readonly property int medium: Math.round(12 * root.r)
        readonly property int large: Math.round(16 * root.r)
        readonly property int extraLarge: Math.round(28 * root.r)
        readonly property int full: 1000
    }

    readonly property QtObject spacing: QtObject {
        readonly property int extraSmall: 4
        readonly property int small: 8
        readonly property int medium: 12
        readonly property int large: 16
        readonly property int extraLarge: 28
    }

    readonly property QtObject padding: QtObject {
        readonly property int extraSmall: 4
        readonly property int small: 8
        readonly property int medium: 12
        readonly property int large: 16
        readonly property int extraLarge: 28
    }

    readonly property QtObject font: QtObject {
        readonly property string sans: Config.appearance.font
        readonly property string mono: "JetBrains Mono"
        readonly property string material: "Material Symbols Rounded"
        readonly property QtObject size: QtObject {
            readonly property int small: 11
            readonly property int smaller: 12
            readonly property int normal: 13
            readonly property int larger: 15
            readonly property int large: 18
            readonly property int extraLarge: 28
        }
    }

    // Material 3 "expressive" motion: cubic-bezier curves for Easing.BezierSpline.
    // Spatial curves overshoot a little (springy); effect curves don't (for opacity and colour).
    readonly property QtObject anim: QtObject {
        readonly property QtObject curves: QtObject {
            readonly property list<real> fastSpatial: [0.42, 1.67, 0.21, 0.90, 1, 1]
            readonly property list<real> spatial: [0.38, 1.21, 0.22, 1.00, 1, 1]
            readonly property list<real> slowSpatial: [0.39, 1.29, 0.35, 0.98, 1, 1]
            readonly property list<real> effects: [0.34, 0.80, 0.34, 1.00, 1, 1]
            readonly property list<real> emphasized: [0.05, 0, 2 / 15, 0.06, 1 / 6, 0.4, 5 / 24, 0.82, 0.25, 1, 1, 1]
            readonly property list<real> standard: [0.2, 0, 0, 1, 1, 1]
        }
        readonly property QtObject durations: QtObject {
            readonly property int fastSpatial: Math.round(350 / root.speed)
            readonly property int spatial: Math.round(500 / root.speed)
            readonly property int slowSpatial: Math.round(650 / root.speed)
            readonly property int effects: Math.round(200 / root.speed)
            readonly property int emphasized: Math.round(500 / root.speed)
            readonly property int standard: Math.round(400 / root.speed)
        }
    }

    readonly property QtObject sizes: QtObject {
        readonly property int barInnerWidth: 40
        readonly property int dashboardWidth: 840
        readonly property int launcherWidth: 600
        readonly property int notificationWidth: 400
        readonly property int sessionButton: 64
        readonly property int osdSliderHeight: 160
        readonly property int popoutWidth: 300
    }
}

lines 11–13Live scaling

Two user options feed the tokens: roundingScale multiplies every corner radius and animationSpeed divides every duration. Because tokens are bindings, moving those settings restyles the whole shell immediately (Settings App 4: The Pages adds the sliders).

        readonly property list<real> spatial: [0.38, 1.21, 0.22, 1.00, 1, 1]
        readonly property list<real> slowSpatial: [0.39, 1.29, 0.35, 0.98, 1, 1]
        readonly property list<real> effects: [0.34, 0.80, 0.34, 1.00, 1, 1]
        readonly property list<real> emphasized: [0.05, 0, 2 / 15, 0.06, 1 / 6, 0.4, 5 / 24, 0.82, 0.25, 1, 1, 1]
        readonly property list<real> standard: [0.2, 0, 0, 1, 1, 1]
    }
    readonly property QtObject durations: QtObject {
        readonly property int fastSpatial: Math.round(350 / root.speed)
        readonly property int spatial: Math.round(500 / root.speed)
        readonly property int slowSpatial: Math.round(650 / root.speed)
        readonly property int effects: Math.round(200 / root.speed)
        readonly property int emphasized: Math.round(500 / root.speed)
        readonly property int standard: Math.round(400 / root.speed)
    }
}

These are the real Material 3 expressive motion curves, with the same numbers Caelestia uses. Each list is a cubic bézier for Easing.BezierSpline: two control points (x1, y1, x2, y2) and the end point (1, 1). A y-value above 1 (like 1.67) means the curve overshoots: a moving panel goes slightly past its target and settles back, which is what makes Caelestia feel springy. Effects curves never overshoot, because opacity above 100% makes no sense. emphasized has two segments (12 numbers) and is used for things leaving the screen.

Module 2: Config#

Tokens are for designers; Config is for users. It reads ~/.config/cae/shell.json, creates it with defaults on the first run, and reloads it whenever you save it.

cae/config/Config.qmlpragma Singleton
import Quickshell
import Quickshell.Io
import QtQuick

// User options, read from ~/.config/cae/shell.json (created with defaults on first run).
// Edit the file while the shell runs: changes apply immediately.
Singleton {
    id: root

    property alias border: adapter.border
    property alias bar: adapter.bar
    property alias dashboard: adapter.dashboard
    property alias launcher: adapter.launcher
    property alias notifs: adapter.notifs
    property alias osd: adapter.osd
    property alias sidebar: adapter.sidebar
    property alias appearance: adapter.appearance
    property alias paths: adapter.paths
    property alias services: adapter.services

    readonly property string dir: (Quickshell.env("XDG_CONFIG_HOME") || Quickshell.env("HOME") + "/.config") + "/cae"

    FileView {
        path: root.dir + "/shell.json"
        watchChanges: true
        onFileChanged: reload()
        onAdapterUpdated: writeAdapter()
        onLoadFailed: err => {
            if (err === FileViewError.FileNotFound) {
                Quickshell.execDetached(["mkdir", "-p", root.dir]);
                writeAdapter();
            }
        }

        JsonAdapter {
            id: adapter

            property JsonObject border: JsonObject {
                property int thickness: 10          // the frame around the screen
                property int rounding: 25           // inner corner radius of the frame
            }
            property JsonObject bar: JsonObject {
                property bool persistent: true      // false = hidden until you hover the left edge
                property list<string> entries: ["logo", "workspaces", "spacer", "activeWindow", "spacer", "tray", "clock", "statusIcons", "power"]
                property int workspacesShown: 5
                property bool workspaceWindows: true
            }
            property JsonObject dashboard: JsonObject {
                property bool showOnHover: true
            }
            property JsonObject launcher: JsonObject {
                property string actionPrefix: ">"
                property int maxShown: 7
            }
            property JsonObject notifs: JsonObject {
                property int defaultExpire: 5000    // ms, when the app doesn't say
            }
            property JsonObject osd: JsonObject {
                property int hideDelay: 2000
            }
            property JsonObject sidebar: JsonObject {
                property bool showOnHover: true
            }
            property JsonObject appearance: JsonObject {
                property bool transparency: false
                property real transparencyBase: 0.85
                property string mode: "dark"        // colour scheme mode for matugen: "dark" | "light"
                property string font: "Rubik"
                property real roundingScale: 1.0    // multiplies every rounding token
                property real animationSpeed: 1.0   // 2 = twice as fast, 0.5 = half speed
            }
            property JsonObject paths: JsonObject {
                property string wallpaperDir: Quickshell.env("HOME") + "/Pictures/Wallpapers"
                property string face: Quickshell.env("HOME") + "/.face"
            }
            property JsonObject services: JsonObject {
                property real audioStep: 0.05
                property real brightnessStep: 0.05
                property string terminal: "kitty"
            }
        }
    }
}

Module 3: Colours#

cae/services/Colours.qmlpragma Singleton
import Quickshell
import Quickshell.Io
import QtQuick
import qs.config

// The Material 3 palette. Colours come from scheme.json (written by matugen from your wallpaper,
// or by the real caelestia CLI). Until a scheme exists, the defaults below (a dark purple) are used.
Singleton {
    id: root

    property string scheme: "default"
    property bool light: false
    readonly property M3Palette palette: M3Palette {}

    // The surface colour of the frame and every drawer, with optional transparency.
    readonly property color surface: Qt.alpha(palette.m3surface, Config.appearance.transparency ? Config.appearance.transparencyBase : 1)

    readonly property string stateDir: (Quickshell.env("XDG_STATE_HOME") || Quickshell.env("HOME") + "/.local/state") + "/cae"

    // A readable foreground for any background colour.
    function on(c: color): color {
        return c.hslLightness < 0.5 ? Qt.hsla(c.hslHue, c.hslSaturation, 0.9, 1) : Qt.hsla(c.hslHue, c.hslSaturation, 0.1, 1);
    }

    function load(text: string): void {
        const s = JSON.parse(text);
        scheme = s.name ?? "dynamic";
        light = s.mode === "light";
        for (const [name, hex] of Object.entries(s.colours ?? {})) {
            const prop = "m3" + name;
            if (palette.hasOwnProperty(prop))
                palette[prop] = "#" + hex.replace("#", "");
        }
    }

    FileView {
        path: root.stateDir + "/scheme.json"
        watchChanges: true
        printErrors: false
        onFileChanged: reload()
        onLoaded: root.load(text())
    }

cae/services/Visibilities.qmlpragma Singleton

// A registry of each screen's ScreenState, so keybinds and IPC can open drawers
// on the monitor you're looking at.

    property var states: ({})

    function register(screen: ShellScreen, state: QtObject): void {

Module 8: Hyprland commands that always work#

Newer Hyprland versions can be configured in Lua, and then dispatcher commands use a different syntax. Caelestia handles both, and so does the rebuild:

cae/services/Hypr.qmlpragma Singleton
import Quickshell
import Quickshell.Hyprland
import QtQuick

// Hyprland commands that work with both config styles. Newer Hyprland can be configured in Lua,
// and then dispatchers use Lua syntax (hl.dsp.focus({ workspace = 3 })) instead of "workspace 3".
// Hyprland.usingLua says which one is active (undefined on older Quickshell = classic).
Singleton {
    readonly property bool lua: Hyprland.usingLua ?? false

    function workspace(id: var): void {
        Hyprland.dispatch(lua ? `hl.dsp.focus({ workspace = "${id}" })` : `workspace ${id}`);
    }
    // relative move: +1 / -1 across existing workspaces
    function workspaceRelative(delta: int): void {
        workspace(delta > 0 ? "r+1" : "r-1");
    }
    function focusWindow(address: string): void {
        Hyprland.dispatch(lua ? `hl.dsp.focus({ window = "address:${address}" })` : `focuswindow address:${address}`);
    }
    function moveWindowToWorkspace(address: string, id: int): void {
        Hyprland.dispatch(lua ? `hl.dsp.window.move({ workspace = ${id}, follow = false, window = "address:${address}" })`
                              : `movetoworkspacesilent ${id},address:${address}`);
    }
}

Module 9: System services#

Four small singletons give the rest of the shell clean access to the system. Each one hides the details (PipeWire objects, sysfs files, /proc parsing) behind a few properties and functions.

cae/services/Audio.qmlpragma Singleton
import Quickshell
import Quickshell.Services.Pipewire
import QtQuick
import qs.config

Singleton {
    id: root

    readonly property PwNode sink: Pipewire.defaultAudioSink
    readonly property PwNode source: Pipewire.defaultAudioSource
    readonly property real volume: sink?.audio?.volume ?? 0
    readonly property bool muted: sink?.audio?.muted ?? false
    readonly property real micVolume: source?.audio?.volume ?? 0
    readonly property bool micMuted: source?.audio?.muted ?? false
    readonly property list<PwNode> sinks: Pipewire.nodes.values.filter(n => n.isSink && !n.isStream && n.audio)
    readonly property string icon: muted ? "volume_off" : volume >= 0.5 ? "volume_up" : volume > 0 ? "volume_down" : "volume_mute"

    function setVolume(v: real): void {
        if (sink?.ready && sink.audio) {
            sink.audio.muted = false;
            sink.audio.volume = Math.max(0, Math.min(1, v));
        }
    }
    function increment(): void { setVolume(volume + Config.services.audioStep) }
    function decrement(): void { setVolume(volume - Config.services.audioStep) }
    function toggleMute(): void { if (sink?.ready) sink.audio.muted = !muted }
    function toggleMicMute(): void { if (source?.ready) source.audio.muted = !micMuted }
    function setSink(node: PwNode): void { Pipewire.preferredDefaultAudioSink = node }

    PwObjectTracker {
        objects: [root.sink, root.source]
    }
}

The default output (sink) and input (source) are tracked with PwObjectTracker so their volume and mute state are live and writable. setVolume clamps to 0–1 and unmutes; increment/decrement use the step from the config; sinks lists every output for the audio popout; icon picks the right volume icon.

cae/services/SysInfo.qmlpragma Singleton
import Quickshell
import Quickshell.Io
import QtQuick

// CPU, memory, temperature and disk usage, read from /proc and /sys (no extra programs).
Singleton {
    id: root

    property real cpu: 0          // 0..1
    property real mem: 0          // 0..1
    property real memUsedGb: 0
    property real memTotalGb: 0
    property real temp: 0         // °C
    property real disk: 0         // 0..1 for /
    property int users: 0
    property string uptime: ""
    property var last: null

    // Only poll while something shows the numbers (the dashboard sets this).
    property int refCount: 0

    Timer {
        running: root.refCount > 0
        interval: 2000
        repeat: true
        triggeredOnStart: true
        onTriggered: {
            stat.reload(); meminfo.reload(); up.reload();
            const f = stat.text().split("\n")[0].trim().split(/\s+/).slice(1).map(Number);
            const idle = f[3] + f[4], total = f.reduce((a, b) => a + b, 0);
            if (root.last)
                root.cpu = 1 - (idle - root.last.idle) / Math.max(1, total - root.last.total);
            root.last = { idle, total };

            const m = meminfo.text();
            const get = k => parseInt((m.match(new RegExp(k + ":\\s+(\\d+)")) ?? [0, 0])[1]);
            const tot = get("MemTotal"), avail = get("MemAvailable");
            root.mem = 1 - avail / Math.max(1, tot);
            root.memUsedGb = (tot - avail) / 1048576;
            root.memTotalGb = tot / 1048576;

            const s = parseFloat(up.text().split(" ")[0]);
            const d = Math.floor(s / 86400), h = Math.floor(s % 86400 / 3600), mi = Math.floor(s % 3600 / 60);
            root.uptime = (d ? d + "d " : "") + (h ? h + "h " : "") + mi + "m";

            if (tempFile.path) {
                tempFile.reload();
                root.temp = parseInt(tempFile.text()) / 1000;
            }
            df.running = true;
        }
    }

    FileView { id: stat; path: "/proc/stat" }
    FileView { id: meminfo; path: "/proc/meminfo" }
    FileView { id: up; path: "/proc/uptime" }
    FileView { id: tempFile; path: "" }

    Process {
        running: true
        command: ["sh", "-c", "for d in /sys/class/hwmon/hwmon*; do case $(cat $d/name) in coretemp|k10temp|zenpower) echo $d/temp1_input; exit;; esac; done"]
        stdout: StdioCollector {
            onStreamFinished: tempFile.path = text.trim()
        }
    }

    Process {
        id: df
        command: ["df", "--output=pcent", "/"]
        stdout: StdioCollector {
            onStreamFinished: root.disk = parseInt(text.split("\n")[1]) / 100
        }
    }
}

CPU (from /proc/stat deltas), memory (MemTotal − MemAvailable), temperature (the CPU's hwmon sensor, found once), disk (df) and uptime. The timer only runs while refCount > 0: the dashboard adds itself when it opens and removes itself when it closes, so these numbers cost nothing while nobody is looking.
