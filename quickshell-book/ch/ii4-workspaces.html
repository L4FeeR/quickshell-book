ii4end-4 4: The Workspace Indicator

ii's workspace indicator packs a lot into 26-pixel cells: occupied workspaces share one continuous background, the active workspace has a pill that trails as it moves, and each workspace shows the icon of its biggest window. Hold Super and the icons turn into numbers. It's worth a chapter of its own because each layer uses a technique you'll reuse elsewhere.

impulse/modules/bar/Workspaces.qmlimport Quickshell
import Quickshell.Hyprland
import Quickshell.Widgets
import QtQuick
import qs.services
import qs.modules.common
import qs.modules.common.widgets
import qs.modules.common.functions

// end-4's workspace indicator: occupied workspaces get one merged background, the active one a
// pill that "trails" (front edge fast, back edge slow), and each workspace shows the icon of its
// window. Holding Super swaps icons for numbers.
Item {
    id: root

    required property ShellScreen screen

    readonly property HyprlandMonitor monitor: Hyprland.monitorFor(screen)
    readonly property int shown: Config.options.bar.workspaces.shown
    readonly property int activeId: monitor?.activeWorkspace?.id ?? 1
    readonly property int group: Math.floor((activeId - 1) / shown)
    readonly property real button: Appearance.sizes.workspaceButton
    readonly property int activeIndex: (activeId - 1) % shown

    function wsAt(index: int): HyprlandWorkspace {
        return Hyprland.workspaces.values.find(w => w.id === group * shown + index + 1) ?? null;
    }
    function occupied(index: int): bool {
        return index >= 0 && index < shown && (wsAt(index)?.toplevels.values.length ?? 0) > 0;
    }

    implicitWidth: button * shown
    implicitHeight: button

    // 1. occupied backgrounds: each over-stretches toward occupied neighbours, so runs merge
    Repeater {
        model: root.shown

        Rectangle {
            required property int index
            readonly property bool occ: root.occupied(index)
            readonly property bool prev: root.occupied(index - 1)
            readonly property bool next: root.occupied(index + 1)
            property real leftEdge: occ ? index * root.button - (prev ? root.button / 2 : 0) : index * root.button + root.button / 2
            property real rightEdge: occ ? (index + 1) * root.button + (next ? root.button / 2 : 0) : index * root.button + root.button / 2

            x: leftEdge
            width: rightEdge - leftEdge
            height: root.button
            radius: root.button / 2
            color: ColorUtils.transparentize(Appearance.colors.colSecondaryContainer, 0.4)

            Behavior on leftEdge { animation: Appearance.animation.elementMoveSmall.numberAnimation.createObject(this) }
            Behavior on rightEdge { animation: Appearance.animation.elementMoveSmall.numberAnimation.createObject(this) }
        }
    }

    // 2. the trailing active indicator
    Rectangle {
        id: active

        readonly property real target: root.activeIndex * root.button + 2
        property real leftEdge: target
        property real rightEdge: target + root.button - 4
        property bool movingRight: false

        onTargetChanged: {
            movingRight = target > leftEdge;
            leftEdge = target;
            rightEdge = target + root.button - 4;
        }

        x: leftEdge
        y: 2
        width: rightEdge - leftEdge
        height: root.button - 4
        radius: height / 2
        color: Appearance.colors.colPrimary

        Behavior on leftEdge {
            NumberAnimation { duration: active.movingRight ? 500 : 250; easing.type: Easing.BezierSpline; easing.bezierCurve: Appearance.animationCurves.expressiveDefaultSpatial }
        }
        Behavior on rightEdge {
            NumberAnimation { duration: active.movingRight ? 250 : 500; easing.type: Easing.BezierSpline; easing.bezierCurve: Appearance.animationCurves.expressiveDefaultSpatial }
        }
    }

    // 3. numbers / app icons on top
    Row {
        Repeater {
            model: root.shown

            Item {
                id: cell

                required property int index
                readonly property int wsId: root.group * root.shown + index + 1
                readonly property bool isActive: index === root.activeIndex
                readonly property HyprlandToplevel biggest: {
                    const ws = root.wsAt(index);
                    if (!ws) return null;
                    const list = ws.toplevels.values;
                    return list.length ? list.reduce((a, b) => ((a.lastIpcObject.size?.[0] ?? 0) * (a.lastIpcObject.size?.[1] ?? 0)
                                                              >= (b.lastIpcObject.size?.[0] ?? 0) * (b.lastIpcObject.size?.[1] ?? 0) ? a : b)) : null;
                }
                readonly property bool showIcon: Config.options.bar.workspaces.showAppIcons && biggest !== null && !GlobalStates.superDown

                width: root.button
                height: root.button

                StyledText {
                    anchors.centerIn: parent
                    opacity: cell.showIcon ? 0 : 1
                    text: cell.wsId
                    font.pixelSize: Appearance.font.pixelSize.smaller
                    color: cell.isActive ? Appearance.colors.colOnPrimary
                         : root.occupied(cell.index) ? Appearance.colors.colOnSecondaryContainer : Appearance.colors.colSubtext
                    Behavior on opacity { animation: Appearance.animation.elementMoveFast.numberAnimation.createObject(this) }
                }

                IconImage {
                    anchors.centerIn: parent
                    implicitSize: cell.showIcon ? root.button * 0.69 : 0
                    opacity: cell.showIcon ? 1 : 0
                    source: Quickshell.iconPath(DesktopEntries.heuristicLookup(cell.biggest?.lastIpcObject.class ?? "")?.icon ?? "", "image-missing")
                    Behavior on implicitSize { animation: Appearance.animation.elementMoveFast.numberAnimation.createObject(this) }
                    Behavior on opacity { animation: Appearance.animation.elementMoveFast.numberAnimation.createObject(this) }
                }

                MouseArea {
                    anchors.fill: parent
                    acceptedButtons: Qt.LeftButton | Qt.RightButton
                    cursorShape: Qt.PointingHandCursor
                    onClicked: mouse => {
                        if (mouse.button === Qt.RightButton) GlobalStates.overviewOpen = !GlobalStates.overviewOpen;
                        else Hypr.workspace(cell.wsId);
                    }
                }
            }
        }
    }

    WheelHandler {
        onWheel: event => Hypr.workspaceRelative(event.angleDelta.y < 0 ? 1 : -1)
    }
}

How it works#

Code walkthrough · piece by piece
lines 18–23Groups of ten
readonly property HyprlandMonitor monitor: Hyprland.monitorFor(screen)
readonly property int shown: Config.options.bar.workspaces.shown
readonly property int activeId: monitor?.activeWorkspace?.id ?? 1
readonly property int group: Math.floor((activeId - 1) / shown)
readonly property real button: Appearance.sizes.workspaceButton
readonly property int activeIndex: (activeId - 1) % shown

shown workspaces form a group (10 by default): 1–10, 11–20… group is the current group number and activeIndex the active workspace's position inside it. All the drawing uses positions inside the group, so moving to workspace 13 shows 11–20 with the pill on the third cell.

lines 25–30Looking up workspaces
function wsAt(index: int): HyprlandWorkspace {
    return Hyprland.workspaces.values.find(w => w.id === group * shown + index + 1) ?? null;
}
function occupied(index: int): bool {
    return index >= 0 && index < shown && (wsAt(index)?.toplevels.values.length ?? 0) > 0;
}

wsAt(i) finds the Hyprland workspace for cell i. occupied(i) also returns false for positions outside the group, which keeps the neighbour checks below simple at both ends.

lines 35–45Occupied: stretch toward neighbours
// 1. occupied backgrounds: each over-stretches toward occupied neighbours, so runs merge

Each cell draws a pill described by a left and a right edge. An occupied cell normally covers its own 26 px. If its left neighbour is occupied too, its left edge moves half a cell further left; same on the right. Two neighbouring pills then overlap by a whole cell, and because they're rounded pills of the same colour, the overlap looks like one long pill. An empty cell collapses to zero width at its centre.

The edges have Behaviors, so when a workspace becomes occupied its pill grows out of its centre and reaches toward its neighbours, and when it empties, the joined pill smoothly splits. The colour is the secondary container at 60%, a soft tint.

Moving right, the right edge (the front) animates in 250 ms and the left edge (the back) in 500 ms; moving left, the opposite. The pill stretches toward the new workspace, then its tail catches up. Same idea as Caelestia's active trail, horizontal.

How the original does it differently#

The original (modules/ii/bar/Workspaces.qml, ~400 lines) uses the same "over-stretch toward occupied neighbours" idea, but goes further with masks:

  1. It draws the occupied pills into an invisible layer and uses a MaskMultiEffect to show one solid background through that mask. Overlaps can never look darker, even with transparency.
  2. It draws the numbers once in the normal colour, then uses a colorizer with a mask: wherever the active pill covers a number, that part of the number is recoloured to on-primary. When the pill slides halfway across a number, the number is literally half one colour and half the other.
  3. A second "interaction" indicator follows the hovered cell, using the same trailing animation.
  4. When a special workspace (scratchpad) is open, the regular workspaces blur and shrink slightly behind it.

These are great next steps once the rebuild's version makes sense; the Effects: Shadows, Blur, Masks, Tints chapter covers MultiEffect masks.

Reuse the trick

"Describe a shape by its edges and animate the edges separately" works for any moving indicator: tab underlines, segmented buttons, list selections. One edge leads, the other follows.

Practice#

Exercise 1Warm-upFive per group

Set bar.workspaces.shown to 5 in config.json and move between workspaces 1–10. What happens at workspace 6?

✅ Solution

The indicator switches to the second group (6–10). Everything is computed from group and activeIndex, so nothing else changes.

Exercise 2PracticeHover indicator

Add a second, transparent pill with a 10% primary tint that follows the hovered cell (use a HoverHandler on the root and compute the cell from point.position.x).

Exercise 3PracticeUrgent workspaces

Colour a workspace's background red when it's urgent (HyprlandWorkspace.urgent).

Exercise 4ChallengeMasked number colours

Implement the original's colorizer: render the numbers in a layer, then use a MultiEffect with maskSource set to the active pill so only the covered part of each number turns on-primary.

💡 Hint

Put the active pill in an item with layer.enabled: true and visible: false; draw the numbers twice (normal colour and on-primary); show the on-primary copy through a MultiEffect { maskEnabled: true; maskSource: pillLayer }.
