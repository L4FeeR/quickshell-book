ly5lyne-dots 5: The Bar

lyne's bar has three groups: left (logo, workspaces, window title), centre (the clock) and right (tray, status icons, bell). It comes in four styles you switch from Settings without restarting, and it can sit at the top or bottom and hide itself. This chapter builds the bar window and every button on it.

The bar window#

lyne/modules/bar/Bar.qmlimport QtQuick
import Quickshell
import Quickshell.Wayland
import qs.config
import qs.services
import qs.components
import qs.modules.quickSettings
import qs.modules.notifications
import qs.modules.dashboard

// One bar per screen. Four looks, chosen with bar.style:
//   "docked"          full width, flush with the screen edge
//   "docked-corners"  the same, with concave corners flowing into the screen
//   "floating"        a rounded bar with a gap around it
//   "islands"         no bar background: each group is its own pill
Scope {
    Variants {
        model: Quickshell.screens

        PanelWindow {
            id: bar
            required property var modelData
            screen: modelData

            anchors {
                top: !Config.barOnBottom
                bottom: Config.barOnBottom
                left: true
                right: true
            }
            // the corners hang below the reserved area, so the window is taller than the bar
            implicitHeight: Config.barReservedHeight + Config.barCornerSize
            exclusiveZone: Config.barAutoHide ? -1 : Config.barReservedHeight
            color: "transparent"
            WlrLayershell.namespace: "lyne_bar"

            // auto-hide: slide away unless hovered or a popup is open
            readonly property bool shown: !Config.barAutoHide || hover.hovered || WindowManagerService.anyModuleOpen
            HoverHandler { id: hover }

            // only the bar takes clicks (the corners click through); when hidden, a 1 px strip
            mask: Region {
                y: Config.barOnBottom ? (bar.shown ? Config.barCornerSize : bar.height - 1) : 0
                width: bar.width
                height: bar.shown ? Config.barReservedHeight : 1
            }

            Item {
                id: slide
                width: parent.width
                height: parent.height
                y: bar.shown ? 0 : (Config.barOnBottom ? height : -height)
                Behavior on y { NumberAnimation { duration: Config.animDurationLong; easing.type: Easing.OutExpo } }

                readonly property real barY: Config.barOnBottom ? Config.barCornerSize : Config.barMargin
                readonly property real cornerY: Config.barOnBottom ? 0 : Config.barHeight

                ConcaveCorner {
                    x: 0
                    y: slide.cornerY
                    size: Config.barCornerSize
                    flipped: Config.barOnBottom
                    color: Config.backgroundTransparentColor
                }
                ConcaveCorner {
                    x: parent.width - width
                    y: slide.cornerY
                    size: Config.barCornerSize
                    flipped: Config.barOnBottom
                    mirrored: true
                    color: Config.backgroundTransparentColor
                }

                Item {
                    id: barArea
                    x: Config.barMargin
                    y: slide.barY
                    width: parent.width - Config.barMargin * 2
                    height: Config.barHeight

                    Rectangle {                              // one background, unless islands
                        anchors.fill: parent
                        visible: !Config.barIslands
                        radius: Config.barFloating ? Config.radiusLarge : 0
                        color: Config.backgroundTransparentColor
                    }

                    BarIsland {                              // left
                        anchors.left: parent.left
                        anchors.leftMargin: Config.spacing
                        anchors.verticalCenter: parent.verticalCenter
                        LauncherButton {}
                        Workspaces {}
                        BarDivider { visible: title.visible }
                        ActiveWindow { id: title }
                    }

                    BarIsland {                              // centre
                        id: centre
                        anchors.centerIn: parent
                        ClockButton { id: clockButton; screenName: bar.modelData.name }
                    }

                    BarIsland {                              // right
                        anchors.right: parent.right
                        anchors.rightMargin: Config.spacing
                        anchors.verticalCenter: parent.verticalCenter
                        TrayWidget {}
                        QuickSettingsButton { id: qsButton; screenName: bar.modelData.name }
                        NotificationButton { id: bellButton; screenName: bar.modelData.name }
                    }
                }
            }

            // the panels that slide out of this bar, one of each per screen
            DashboardWindow { screen: bar.modelData; screenName: bar.modelData.name; barWindow: bar; anchorItem: clockButton }
            QuickSettingsWindow { screen: bar.modelData; screenName: bar.modelData.name; barWindow: bar; anchorItem: qsButton }
            NotificationHistory { screen: bar.modelData; screenName: bar.modelData.name; barWindow: bar; anchorItem: bellButton }
        }
    }
}

The window is the bar's height plus its margin plus room for the concave corners, which hang below the reserved area. exclusiveZone tells the compositor how much space to keep free for the bar: the bar and margin, not the corners. An auto-hiding bar reserves nothing (-1, the same value ExclusionMode.Ignore sends). The namespace lets Hyprland blur behind it (lyne-dots 12: Assembling the Shell and Configuring Hyprland).

The bar shows when auto-hide is off, while the pointer is over it, or while any popup is open (WindowManagerService.anyModuleOpen, lyne-dots 6: Panels That Grow Out of the Bar), so it doesn't slide away under the Quick Settings panel you're using.

A transparent window still swallows clicks everywhere it covers. The Region mask limits input to the bar itself, so the corners hanging below it click through to your windows. The logo button#

lyne/components/LauncherButton.qmlimport QtQuick
import qs.config
import qs.services

// The logo at the left of the bar. Click: launcher. Right click: the
// launcher's actions (">"). Middle click: clipboard history (";").
// Hovering makes the dot hop: crouch, leap stretched, land flat, settle.
BarButton {
    id: root

    active: LauncherService.visible
    onClicked: LauncherService.toggle()
    onRightClicked: LauncherService.toggleMode(">")
    onMiddleClicked: LauncherService.toggleMode(";")
    onHoveredChanged: if (hovered && !hop.running) hop.start()

    LyneMark {
        id: mark
        height: Config.fontSizeIcon * 0.8
        width: height * 1.2
        // the dot takes the accent colour; the line is derived from it
        dotColor: Config.accentColor
    }

    SequentialAnimation {
        id: hop
        // anticipation: squash down
        ParallelAnimation {
            NumberAnimation { target: mark; property: "stretch"; to: -0.3; duration: 90; easing.type: Easing.OutQuad }
        }
        // leap: rise, stretched tall
        ParallelAnimation {
            NumberAnimation { target: mark; property: "lift"; to: 26; duration: 220; easing.type: Easing.OutQuad }
            NumberAnimation { target: mark; property: "stretch"; to: 0.25; duration: 120 }
        }
        // fall back down
        ParallelAnimation {
            NumberAnimation { target: mark; property: "lift"; to: 0; duration: 200; easing.type: Easing.InQuad }
            NumberAnimation { target: mark; property: "stretch"; to: 0.15; duration: 200 }
        }
        // impact: flat, then back to round
        NumberAnimation { target: mark; property: "stretch"; to: -0.35; duration: 70 }
        NumberAnimation { target: mark; property: "stretch"; to: 0; duration: 260; easing.type: Easing.OutBack; easing.overshoot: 3 }
    }
}

Workspaces#

lyne/components/Workspaces.qmlimport QtQuick
import Quickshell
import Quickshell.Hyprland
import qs.config
import qs.services

// The workspace strip. One slot per workspace (1 … count); the active slot
// is wider and an accent "indicator" pill slides to it. The indicator's two
// edges move at different speeds: the edge in the direction of travel leads,
// the other follows, so the pill stretches and snaps like elastic.
//
// Three looks (bar.workspaces.style): "pills" (a bar per workspace),
// "dots" (small circles) and "numbers".
Item {
    id: root

    property int count: Config.barWorkspaceCount
    property string style: Config.barWorkspaceStyle

    readonly property int activeId: Hyprland.focusedWorkspace?.id ?? 1
    readonly property int activeIndex: activeId - 1

    readonly property int itemWidth: style === "numbers" ? Config.fontSizeSmall + Config.padding * 2 : Config.fontSizeSmall + Math.round(Config.padding / 2)
    readonly property int activeWidth: style === "numbers" ? itemWidth : itemWidth * 2
    readonly property int itemSpacing: Math.round(Config.padding * 2 / 3)
    readonly property int indicatorHeight: Config.fontSizeSmall + Config.padding

    function isOccupied(id) { return Hyprland.workspaces.values.some(w => w.id === id && w.toplevels.values.length > 0); }
    function isUrgent(id) { return Hyprland.workspaces.values.some(w => w.id === id && w.urgent); }

    // where each slot starts and how wide it is, from left to right
    function slotX(index) {
        let x = 0;
        for (let i = 0; i < index; i++)
            x += (i === activeIndex ? activeWidth : itemWidth) + itemSpacing;
        return x;
    }
    function slotWidth(index) { return index === activeIndex ? activeWidth : itemWidth; }

    implicitWidth: slotX(count) - itemSpacing
    implicitHeight: Config.barButtonHeight

    // ── The indicator ─────────────────────────────────────────────────────
    Rectangle {
        id: indicator
        property real leftEdge: root.slotX(root.activeIndex)
        property real rightEdge: leftEdge + root.slotWidth(root.activeIndex)
        property int direction: 0                     // +1 moving right, -1 moving left

        z: root.style === "numbers" ? 0 : 1          // numbers are drawn on top of it
        x: leftEdge
        width: rightEdge - leftEdge
        height: root.indicatorHeight
        anchors.verticalCenter: parent.verticalCenter
        radius: root.style === "dots" ? height / 2 : Config.radius
        color: Config.accentColor
        visible: root.activeIndex >= 0 && root.activeIndex < root.count

        // the leading edge is quick, the trailing edge slower: the stretch
        Behavior on leftEdge { NumberAnimation { duration: indicator.direction < 0 ? Config.animDuration * 0.6 : Config.animDuration; easing.type: Easing.OutCubic } }
        Behavior on rightEdge { NumberAnimation { duration: indicator.direction > 0 ? Config.animDuration * 0.6 : Config.animDuration; easing.type: Easing.OutCubic } }
    }
    property int lastIndex: activeIndex
    onActiveIndexChanged: { indicator.direction = Math.sign(activeIndex - lastIndex); lastIndex = activeIndex; }
