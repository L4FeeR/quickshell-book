ly6lyne-dots 6: Panels That Grow Out of the Bar

Click the clock and the dashboard doesn't just appear: it slides out from under the bar, its top corners curve into the bar's edge, and the bar and panel look like one surface. The Quick Settings panel and the notification history do the same. This chapter builds that effect once, as two reusable components, then uses it for Quick Settings and the dashboard. It also adds the rule that only one popup is open at a time.

One popup at a time#

lyne/services/WindowManagerService.qmlpragma Singleton
import QtQuick
import Quickshell
import qs.services

// Keeps the shell's popups polite: only one is open at a time, and the bar
// knows when something is open (an auto-hiding bar stays out while it is).
Singleton {
    id: root

    // names of the popups open right now: { "Launcher": true, … }
    property var openModules: ({})
    readonly property bool anyModuleOpen: Object.keys(openModules).length > 0

    // every popup listens to this and closes unless it is `except`
    signal closeRequested(string except)

    function registerOpen(name) {
        if (openModules[name])
            return;
        const copy = Object.assign({}, openModules);   // a new object, so bindings notice
        copy[name] = true;
        openModules = copy;
    }
    function registerClose(name) {
        if (!openModules[name])
            return;
        const copy = Object.assign({}, openModules);
        delete copy[name];
        openModules = copy;
    }

    // Close everything else, then run `open` (which may also close `name`
    // again: that's how a shortcut toggles its popup).
    function openExclusive(name, open) {
        closeRequested(name);
        open();
    }
}

The attached panel#

lyne/components/AttachedPanel.qmlimport QtQuick
import qs.config

// lyne's signature panel: it grows OUT of the bar. It is flush with the bar
// edge, the corners where it meets the bar are curved inward (fillets), and
// it slides in from behind the bar instead of fading in the middle of nowhere.
//
//      ═══════════ bar ═══════════
//           ◟ ┌────────┐ ◞      <- fillets join the panel to the bar
//             │ panel  │
//             └────────┘
//
// Give it a size and put content inside:
//   AttachedPanel { width: 300; height: 200; shown: open; edge: "top"; … }
Item {
    id: root

    property bool shown: false
    property string edge: "top"                        // the side the bar is on: "top" | "bottom"
    property color color: Config.backgroundTransparentColor
    property real radius: Config.radiusLarge
    property real filletSize: Config.radiusLarge

    default property alias content: body.data
    readonly property alias body: body
    readonly property bool onTop: edge === "top"

    // 0 = hidden behind the bar, 1 = fully out
    property real progress: 0
    // Start at 0 and only then follow `shown`, so the panel animates in even
    // when it is created already shown.
    property bool _ready: false
    Component.onCompleted: Qt.callLater(() => root._ready = true)
    on_ReadyChanged: progress = Qt.binding(() => root.shown && root._ready ? 1 : 0)
    Behavior on progress {
        NumberAnimation {
            duration: root.shown ? Config.animDurationLong : Config.animDuration
            easing.type: Easing.OutExpo
        }
    }

    // The clipper is the panel's area plus room for a fillet on each side.
    // Whatever slides past its edge (towards the bar) is cut off, so the
    // panel seems to come out from under the bar.
    Item {
        id: clipper
        x: -root.filletSize
        width: root.width + root.filletSize * 2
        height: root.height
        clip: true

        Item {
            id: slide
            x: root.filletSize
            width: root.width
            height: root.height
            y: (1 - root.progress) * (root.onTop ? -root.height : root.height)
            opacity: Math.min(1, root.progress * 2)

            // The body: rounded, except the two corners touching the bar.
            // A rounded rectangle taller than the panel, pushed past the bar
            // edge and clipped there, leaves square corners on that side.
            // (Two overlapping see-through rectangles would show a darker
            // strip where they overlap; this is one shape.)
            Item {
                anchors.fill: parent
                clip: true
                Rectangle {
                    width: parent.width
                    height: parent.height + root.radius
                    y: root.onTop ? -root.radius : 0
                    radius: root.radius
                    color: root.color
                }
            }

            // the fillets, at the bar edge, just outside the panel's sides
            ConcaveCorner {
                x: -root.filletSize
                y: root.onTop ? 0 : parent.height - root.filletSize
                size: root.filletSize
                color: root.color
                mirrored: true
                flipped: !root.onTop
            }
            ConcaveCorner {
                x: parent.width
                y: root.onTop ? 0 : parent.height - root.filletSize
                size: root.filletSize
                color: root.color
                flipped: !root.onTop
            }

            Item {
                id: body
                anchors.fill: parent
            }
        }
    }
}

The popup window around it#

A panel needs a window. It can't live inside the bar's window, which is only as tall as the bar.

lyne/components/AttachedPopup.qmlimport QtQuick
import Quickshell
import Quickshell.Wayland
import Quickshell.Hyprland
import qs.config

// A popup window that holds one AttachedPanel right under (or above) the bar.
// The Quick Settings panel, the dashboard and the notification history all
// use it. Only the panel takes clicks; the rest of the window lets them
// through. Clicking anywhere else closes it.
PanelWindow {
    id: win

    property bool shown: false
    property Item anchorItem: null              // the bar button it opens from: the panel centres under it
    property real panelX: width / 2             // where the panel's centre is, in screen x
    property real panelWidth: 300
    property real panelHeight: 200
    property var barWindow: null                // clicks on the bar don't count as "outside"
    default property alias content: holder.data
    signal dismissed                            // Escape or a click outside

    // measure the button's position when opening (the bar spans the whole
    // screen, so its window x is the screen x)
    onShownChanged: if (shown && anchorItem) panelX = anchorItem.mapToItem(null, anchorItem.width / 2, 0).x

    // stays mapped until the slide-out animation has finished
    visible: shown || panel.progress > 0

    anchors {
        top: !Config.barOnBottom
        bottom: Config.barOnBottom
        left: true
        right: true
    }
    exclusionMode: ExclusionMode.Ignore         // start at the screen edge, under the bar
    implicitHeight: Config.barReservedHeight + panelHeight + 4
    color: "transparent"
    WlrLayershell.layer: WlrLayer.Top
    WlrLayershell.namespace: "lyne_attached"
    WlrLayershell.keyboardFocus: shown ? WlrKeyboardFocus.OnDemand : WlrKeyboardFocus.None

    mask: Region { item: panel }

    HyprlandFocusGrab {
        active: win.shown
        windows: win.barWindow ? [win, win.barWindow] : [win]
        onCleared: win.dismissed()
    }

    AttachedPanel {
        id: panel
        shown: win.shown
        edge: Config.barOnBottom ? "bottom" : "top"
        width: win.panelWidth
        height: win.panelHeight
        x: Math.max(Config.spacing + filletSize, Math.min(win.width - width - Config.spacing - filletSize, win.panelX - width / 2))
        y: Config.barOnBottom ? win.height - Config.barReservedHeight - height : Config.barReservedHeight

        Item {
            id: holder
            anchors.fill: parent
            focus: true
            Keys.onEscapePressed: win.dismissed()
        }
    }
}

Quick Settings#

Each popup has a service (state and logic) and a window (drawing). The service:

lyne/services/QuickSettingsService.qmlpragma Singleton
import QtQuick
import Quickshell
import qs.services
import Quickshell.Io
import Quickshell.Hyprland

// Which screen's Quick Settings panel is open, and on which page.
Singleton {
    id: root

    property bool visible: false
    property string screenName: ""
    property string page: "main"              // "main" | "wifi" | "bluetooth" | "sound"

    function toggle(screen) {
        if (visible && screenName === screen) { close(); return; }
        WindowManagerService.openExclusive("QuickSettings", () => {
            screenName = screen;
            page = "main";
            visible = true;
        });
    }
    function close() { visible = false; }
    function openPage(p) { page = p; }

    Connections {
        target: WindowManagerService
        function onCloseRequested(except) { if (except !== "QuickSettings") root.close(); }
    }
    onVisibleChanged: visible ? WindowManagerService.registerOpen("QuickSettings") : WindowManagerService.registerClose("QuickSettings")

    IpcHandler {
        target: "quicksettings"
        function toggle(): void { root.toggle(Hyprland.focusedMonitor?.name ?? Quickshell.screens[0].name); }
    }
}

Singleton {

    // toggle("media", "DP-1"): opens on that tab; again on the same tab closes
    function toggle(t, screen) {
        if (visible && screenName === screen && (t === "" || t === tab)) { close(); return; }
        WindowManagerService.openExclusive("Dashboard", () => {
            screenName = screen;
            if (t !== "")
                tab = t;
            visible = true;
        });
    }
    function close() { visible = false; }

    Connections {
        target: WindowManagerService
        function onCloseRequested(except) { if (except !== "Dashboard") root.close(); }
    }
    onVisibleChanged: visible ? WindowManagerService.registerOpen("Dashboard") : WindowManagerService.registerClose("Dashboard")

    IpcHandler {
        target: "dashboard"
        function toggle(tab: string): void { root.toggle(tab, root.focusedScreen() || Quickshell.screens[0].name); }
    }
}
