ly3lyne-dots 3: Design Tokens, Themes and the Colour Fade

Open any widget in lyne and you won't find "#7aa2f7" or radius: 12. You'll find Config.accentColor and Config.radius. These names are design tokens: one place decides what "the accent" or "a normal corner" means, and every widget follows. This chapter builds the three files behind them: the theme files, the theme service that loads and exports them, and Config, which turns settings and themes into tokens and fades between themes.

Section Used for

The palette roles:

Role Meaning
background panels and the bar
surface0 … surface3 raised layers, lightest last in a dark theme: cards, hover, borders, inactive items
text, subtext, muted normal text, secondary text, hints and empty states
textReverse text on the accent colour
accent the one highlight colour: active workspace, selection, sliders
success, warning, error charging, caution, critical

Roles describe purpose, not looks. Widgets ask for "the colour for secondary text", so any palette that fills the roles works everywhere (Theming 1: Colours, Palettes and Roles explains the idea in depth).

Mixing colours properly#

Fading from Tokyo Night's blue accent to Gruvbox's orange in plain RGB passes through a dull grey-brown in the middle. lyne fades in OKLab, a colour space designed so that equal steps look like equal changes. The maths is a few matrix multiplications:

lyne/services/color.js// Colour maths for theme fades. Mixing two colours halfway in plain RGB goes
// through muddy greys (blue + yellow = grey). OKLab is a colour space where
// a straight line between two colours looks like an even change to our eyes,
// so the fade from one theme to the next stays clean.
.pragma library

function toLinear(c) {
    return c <= 0.04045 ? c / 12.92 : Math.pow((c + 0.055) / 1.055, 2.4);
}
function toGamma(c) {
    return c <= 0.0031308 ? c * 12.92 : 1.055 * Math.pow(c, 1 / 2.4) - 0.055;
}

// a Qt color (r, g, b, a from 0 to 1) -> [L, a, b, alpha]
function toOklab(color) {
    const r = toLinear(color.r), g = toLinear(color.g), b = toLinear(color.b);
    const l = Math.cbrt(0.4122214708 * r + 0.5363325363 * g + 0.0514459929 * b);
    const m = Math.cbrt(0.2119034982 * r + 0.6806995451 * g + 0.1073969566 * b);
    const s = Math.cbrt(0.0883024619 * r + 0.2817188376 * g + 0.6299787005 * b);
    return [
        0.2104542553 * l + 0.7936177850 * m - 0.0040720468 * s,
        1.9779984951 * l - 2.4285922050 * m + 0.4505937099 * s,
        0.0259040371 * l + 0.7827717662 * m - 0.8086757660 * s,
        color.a
    ];
}

// [L, a, b, alpha] -> [r, g, b, alpha], each 0..1
function fromOklab(lab) {
    const l = Math.pow(lab[0] + 0.3963377774 * lab[1] + 0.2158037573 * lab[2], 3);
    const m = Math.pow(lab[0] - 0.1055613458 * lab[1] - 0.0638541728 * lab[2], 3);
    const s = Math.pow(lab[0] - 0.0894841775 * lab[1] - 1.2914855480 * lab[2], 3);
    const clamp = x => Math.max(0, Math.min(1, x));
    return [
        clamp(toGamma(4.0767416621 * l - 3.3077115913 * m + 0.2309699292 * s)),
        clamp(toGamma(-1.2684380046 * l + 2.6097574011 * m - 0.3413193965 * s)),
        clamp(toGamma(-0.0041960863 * l - 0.7034186147 * m + 1.7076147010 * s)),
        lab[3]
    ];
}

// the colour a fraction t (0..1) of the way from one OKLab colour to another
function mix(fromLab, toLab, t) {
    return fromOklab(fromLab.map((v, i) => v + (toLab[i] - v) * t));
}

.pragma library shares one copy of this file between every QML file that imports it, and keeps it free of any QML context. It's pure JavaScript.

lines 7–12Undoing the screen's curve
function toLinear(c) {

Colour values on screen aren't proportional to light: 0.5 is much darker than half of 1.0. toLinear converts a channel to real light intensity, toGamma converts back. These are the standard sRGB formulas.

lines 14–26To OKLab

Linear RGB goes through two fixed matrices with a cube root in between (from the OKLab specification). The result is L (lightness), a (green ↔ red) and b (blue ↔ yellow), plus the alpha kept as is.

The same steps in reverse. clamp keeps each channel between 0 and 1, because a mix can land just outside what a screen can show.

A straight line between two OKLab colours: each component moves the fraction t of the way. t = 0 is the start colour, t = 1 the end.

The theme service#

lyne/services/ThemeService.qmlpragma Singleton
import QtQuick
import Quickshell
import qs.services
import Quickshell.Io

// Loads the theme files, publishes the chosen palette, and paints the rest of
// the desktop with it: kitty, Hyprland's borders and GTK's light/dark mode.
// A theme file is JSON in lyne-dots' format (themes/*.json shows it), so the
// themes you make for the real lyne-dots work here too, and the other way round.
Singleton {
    id: root

    // built-in themes ship with the shell; your own go in ~/.local/themes
    readonly property string builtinDir: Quickshell.shellPath("themes")
    readonly property string userDir: Quickshell.env("HOME") + "/.local/themes"
    readonly property string kittyPath: Quickshell.env("HOME") + "/.config/kitty/current-theme.conf"

    // slug ("tokyonight") -> parsed file
    property var themes: ({})
    // [{ slug, name, variant, palette }] sorted by name, for pickers
    readonly property var list: Object.keys(themes).sort((a, b) => themes[a].name.localeCompare(themes[b].name))
        .map(slug => ({ slug: slug, name: themes[slug].name, variant: themes[slug].variant ?? "dark", palette: themes[slug].palette }))

    readonly property string currentName: StateService.get("theme.name", "tokyonight")
    readonly property var current: themes[currentName] ?? null
    readonly property bool isDark: (current?.variant ?? "dark") === "dark"

    // What Config fades to. Only replaced when the colours really change.
    property var palette: ({})

    // Start the shell with LYNE_NO_EXPORT=1 to try it without touching kitty,
    // Hyprland or GTK (on another desktop, for example).
    readonly property bool exportEnabled: !Quickshell.env("LYNE_NO_EXPORT")

    function set(slug) {
        if (themes[slug])
            StateService.set("theme.name", slug);
    }
    function step(delta) {
        const slugs = list.map(t => t.slug);
        const i = slugs.indexOf(currentName);
        set(slugs[(i + delta + slugs.length) % slugs.length]);
    }

    onCurrentChanged: {
        if (!current)
            return;
        palette = current.palette;
        if (!exportEnabled)
            return;
        _applyKitty(current.terminal);
        _applyHyprland(current.hyprland);
        _applyGtk();
    }

    // ── Loading every theme file with one process ─────────────────────────
    // It prints each file's name, its contents, then a line with the ASCII
    // record separator (\036), which never appears in JSON.
    function reload() { loader.running = true; }
    Component.onCompleted: reload()

    Process {
        id: loader
        command: ["sh", "-c", 'for f in "$1"/*.json "$2"/*.json; do [ -f "$f" ] || continue; basename "$f" .json; cat "$f"; printf "\\n\\036\\n"; done',
            "sh", root.builtinDir, root.userDir]
        stdout: StdioCollector {
            onStreamFinished: {
                const out = {};
                for (const chunk of text.split("\x1e")) {
                    const nl = chunk.trim().indexOf("\n");
                    if (nl < 0)
                        continue;
                    const slug = chunk.trim().slice(0, nl);
                    try {
                        const data = JSON.parse(chunk.trim().slice(nl + 1));
                        if (data.palette)
                            out[slug] = data;           // later files (yours) replace built-ins
                    } catch (e) {
                        console.warn("[Theme] skipping broken theme", slug, e);
                    }
                }
                root.themes = out;
            }
        }
    }

    // ── Painting the desktop ────────────────────────────────────────────────
    // Writes text to a file through `sh`. The path and the text are passed
    // as arguments, never pasted into the command, so quotes can't break it.
    function _write(path, text, after) {
        Quickshell.execDetached(["sh", "-c", 'mkdir -p "$(dirname "$1")" && printf "%s" "$2" > "$1"; ' + (after ?? ""), "sh", path, text]);
    }

    // kitty: `include current-theme.conf` in kitty.conf, then SIGUSR1 reloads every window
    function _applyKitty(t) {
        if (!t)
            return;
        const keys = {
            background: "background", foreground: "foreground", cursor: "cursor", cursorTextColor: "cursor_text_color",
            selectionBackground: "selection_background", selectionForeground: "selection_foreground", urlColor: "url_color",
            activeTabBackground: "active_tab_background", activeTabForeground: "active_tab_foreground",
            inactiveTabBackground: "inactive_tab_background", inactiveTabForeground: "inactive_tab_foreground",
            activeBorderColor: "active_border_color", inactiveBorderColor: "inactive_border_color"
        };
        const lines = ["# written by the lyne shell: do not edit"];
        for (const k in keys)
            if (t[k]) lines.push(keys[k] + " " + t[k]);
        for (let i = 0; i < 18; i++)
            if (t["color" + i]) lines.push("color" + i + " " + t["color" + i]);
        _write(kittyPath, lines.join("\n") + "\n", "pkill -USR1 -x kitty; true");
    }

    // Hyprland: borders and shadow, applied now. A Lua config (hyprland.lua)
    // takes `hyprctl eval` with hl.config(); a classic config takes `keyword`.
    function _applyHyprland(h) {
        if (!h)
            return;
        const lua = `hl.config({ general = { col = { active_border = "rgba(${h.activeBorder})", inactive_border = "rgba(${h.inactiveBorder})" } },`
            + ` decoration = { shadow = { color = "rgba(${h.shadowColor})" } } })`;
        const classic = `keyword general:col.active_border rgba(${h.activeBorder}); keyword general:col.inactive_border rgba(${h.inactiveBorder});`
            + ` keyword decoration:shadow:color rgba(${h.shadowColor})`;
        Quickshell.execDetached(["sh", "-c", 'if [ -f "$HOME/.config/hypr/hyprland.lua" ]; then hyprctl eval "$1"; else hyprctl --batch "$2"; fi',
            "sh", lua, classic]);
    }

    // GTK 4 / libadwaita apps, Firefox and websites follow this preference
    function _applyGtk() {
        Quickshell.execDetached(["gsettings", "set", "org.gnome.desktop.interface", "color-scheme", isDark ? "prefer-dark" : "prefer-light"]);
    }

    IpcHandler {
        target: "theme"
        function set(slug: string): void { root.set(slug); }
        function next(): void { root.step(1); }
        function previous(): void { root.step(-1); }
        function get(): string { return root.currentName; }
        function list(): string { return root.list.map(t => t.slug).join("\n"); }
        function reload(): void { root.reload(); }
    }
}

Nerd Fonts put icons in the "private use" area of Unicode. "\u{f057e}" is the volume icon (named nf-md-volume_high on nerdfonts.com/cheat-sheet). lyne writes these codes directly in its widgets; the rebuild names them once here so code reads Icons.volumeHigh. To add an icon, search the cheat sheet, copy its code, and add a line. wifi is a list from no signal to full, so a widget can pick Icons.wifi[strength].
