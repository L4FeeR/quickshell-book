ii6end-4 6: The Overview and Search

Press Super+Tab (or right-click the workspaces, or just tap Super) and ii shows the overview: a search box above a grid of your workspaces, each a small copy of the screen with live previews of its windows. Start typing and the grid turns into launcher results: apps, a calculator, commands and web search. Drag a window onto another workspace to move it.

modules/overview/  Overview  OverviewGrid  AppIconImage
services/          LauncherSearch  HyprlandData

Module 1: Window data#

The overview needs every window's position and size on screen, which Quickshell's toplevel list doesn't include. ii asks Hyprland directly.

impulse/services/HyprlandData.qmlpragma Singleton
import Quickshell
import Quickshell.Io
import Quickshell.Hyprland
import QtQuick

// Window positions and sizes for the overview, from `hyprctl clients -j`, refreshed on every
// Hyprland event that could move a window.
Singleton {
    id: root

    property var windowList: []
    property var monitors: []

    function refresh(): void {
        clients.running = true;
        mons.running = true;
    }

    Component.onCompleted: refresh()

    Connections {
        target: Hyprland
        function onRawEvent(event: HyprlandEvent): void {
            if (["openwindow", "closewindow", "movewindow", "movewindowv2", "windowtitle", "workspace", "changefloatingmode", "fullscreen"].includes(event.name))
                root.refresh();
        }
    }

    Process {
        id: clients
        command: ["hyprctl", "clients", "-j"]
        stdout: StdioCollector { onStreamFinished: root.windowList = JSON.parse(text) }
    }
    Process {
        id: mons
        command: ["hyprctl", "monitors", "-j"]
        stdout: StdioCollector { onStreamFinished: root.monitors = JSON.parse(text) }
    }
}

lines 12–13The data

property var monitors: []

windowList is hyprctl clients -j parsed: one object per window with address, at (x, y), size (w, h), workspace, class, title… monitors is hyprctl monitors -j, used to turn global positions into positions on one monitor.

lines 15–18Refresh
function refresh(): void {
    clients.running = true;
    mons.running = true;
}

Both commands run again whenever something might have moved.

lines 22–28Listening to Hyprland
Connections {
    target: Hyprland
    function onRawEvent(event: HyprlandEvent): void {
        if (["openwindow", "closewindow", "movewindow", "movewindowv2", "windowtitle", "workspace", "changefloatingmode", "fullscreen"].includes(event.name))
            root.refresh();
    }
}

rawEvent delivers every event from Hyprland's event socket. Only events that change window positions trigger a refresh, so the data stays current without polling.

Module 2: Search#

impulse/services/LauncherSearch.qmlpragma Singleton
import Quickshell
import Quickshell.Io
import QtQuick

// What the overview's search box shows for a query: apps, a math result (qalc),
// and fallbacks to run the text as a command or search the web.
Singleton {
    id: root

    property string query: ""
    property string mathResult: ""

    readonly property var results: {
        const q = query.trim();
        if (!q) return [];
        const out = [];
        if (mathResult) out.push({ type: "Math", name: mathResult, icon: "calculate", run: () => Quickshell.execDetached(["wl-copy", mathResult]) });
        const lq = q.toLowerCase();
        for (const app of DesktopEntries.applications.values) {
            if (out.length > 8) break;
            if (!app.noDisplay && (app.name.toLowerCase().includes(lq) || (app.genericName ?? "").toLowerCase().includes(lq)))
                out.push({ type: "App", name: app.name, iconName: app.icon, run: () => app.execute() });
        }
        out.push({ type: "Command", name: q, icon: "terminal", run: () => Quickshell.execDetached(["sh", "-c", q]) });
        out.push({ type: "Web", name: "Search the web for " + q, icon: "travel_explore", run: () => Qt.openUrlExternally("https://duckduckgo.com/?q=" + encodeURIComponent(q)) });
        return out;
    }

    onQueryChanged: {
        mathResult = "";
        if (/^[\d\s+\-*/^().%]+$|\bto\b/.test(query.trim())) math.running = true;
    }

    Process {
        id: math
        command: ["qalc", "-t", root.query]
        stdout: StdioCollector { onStreamFinished: root.mathResult = text.trim() }
    }
}

A maths result first (if any), then up to ~9 matching apps, then two fallbacks that always appear: run the text as a shell command, or search the web for it. Each result has a type label, a name, an icon (Material symbol name or app icon name) and a run function.

lines 30–33When to calculate
onQueryChanged: {
    mathResult = "";
    if (/^[\d\s+\-*/^().%]+$|\bto\b/.test(query.trim())) math.running = true;
}

Only text made of digits and operators, or containing " to " (unit/currency conversion), is sent to qalc. Every other query skips the calculator.

lines 35–39qalc
Process {
    id: math
    command: ["qalc", "-t", root.query]
    stdout: StdioCollector { onStreamFinished: root.mathResult = text.trim() }
}

Runs with the query and stores the terse result.

In the original

services/LauncherSearch.qml supports prefixes (/ actions, : emoji, ; clipboard, = maths, ! web…), fuzzy app search, and Hyprland-aware actions. Results render as a SearchWidget with keyboard navigation, and the same window hosts the emoji and clipboard pickers.
