ii3end-4 3: The Bar

ii's bar is a full-width strip at the top with three rounded groups in the middle, a sidebar button and the window title on the left, and indicators and the tray on the right. In its default "hug" style, two concave corners hang under its ends so the bar seems to pour into the screen's rounded corners.

Module 1: The bar window#

impulse/modules/bar/Bar.qmlimport Quickshell
import Quickshell.Wayland
import QtQuick
import qs.services
import qs.modules.common
import qs.modules.common.widgets

// The top bar window, one per monitor. With cornerStyle 0 ("hug") two concave corners hang
// under the bar so it melts into the screen; with auto-hide it slides away until you hover.
Scope {
    Variants {
        model: Quickshell.screens

        PanelWindow {
            id: barRoot

            required property ShellScreen modelData
            readonly property bool autoHide: Config.options.bar.autoHide.enable
            readonly property bool mustShow: hover.containsMouse || GlobalStates.superDown
            readonly property bool hug: Config.options.bar.cornerStyle === 0
            readonly property bool floating: Config.options.bar.cornerStyle === 1
            readonly property real gap: floating ? Appearance.sizes.hyprlandGapsOut : 0

            screen: modelData
            visible: GlobalStates.barOpen && !GlobalStates.screenLocked
            anchors {
                top: !Config.options.bar.bottom
                bottom: Config.options.bar.bottom
                left: true
                right: true
            }
            implicitHeight: Appearance.sizes.barHeight + gap * 2 + Appearance.rounding.screenRounding
            exclusiveZone: autoHide && !mustShow ? 0 : Appearance.sizes.barHeight + gap
            color: "transparent"
            mask: Region { item: hoverArea }
            WlrLayershell.namespace: "quickshell:bar"

            MouseArea {
                id: hover
                anchors.fill: parent
                hoverEnabled: true
                acceptedButtons: Qt.NoButton

                // the part that takes input: the bar itself, or a thin strip while hidden
                Item {
                    id: hoverArea
                    anchors.left: parent.left
                    anchors.right: parent.right
                    y: Config.options.bar.bottom ? parent.height - height : 0
                    height: barRoot.autoHide && !barRoot.mustShow ? 2 : Appearance.sizes.barHeight + barRoot.gap * 2
                }

                Item {
                    id: slide
                    anchors.left: parent.left
                    anchors.right: parent.right
                    height: Appearance.sizes.barHeight + barRoot.gap * 2 + Appearance.rounding.screenRounding
                    // slide out of view when auto-hidden
                    y: (Config.options.bar.bottom ? parent.height - height : 0)
                       + (barRoot.autoHide && !barRoot.mustShow ? (Config.options.bar.bottom ? height : -height) : 0)

                    Behavior on y {
                        animation: Appearance.animation.elementMoveFast.numberAnimation.createObject(this)
                    }

                    BarContent {
                        id: content
                        screen: barRoot.modelData
                        anchors.left: parent.left
                        anchors.right: parent.right
                        anchors.margins: barRoot.gap
                        y: Config.options.bar.bottom ? parent.height - height - barRoot.gap : barRoot.gap
                        height: Appearance.sizes.barHeight
                        radius: barRoot.floating ? Appearance.rounding.normal : 0
                    }

                    RoundCorner {
                        visible: barRoot.hug
                        size: Appearance.rounding.screenRounding
                        color: Appearance.colors.colLayer0
                        corner: Config.options.bar.bottom ? RoundCorner.CornerEnum.BottomLeft : RoundCorner.CornerEnum.TopLeft
                        anchors.left: parent.left
                        y: Config.options.bar.bottom ? content.y - size : content.y + content.height
                    }
                    RoundCorner {
                        visible: barRoot.hug
                        size: Appearance.rounding.screenRounding
                        color: Appearance.colors.colLayer0
                        corner: Config.options.bar.bottom ? RoundCorner.CornerEnum.BottomRight : RoundCorner.CornerEnum.TopRight
                        anchors.right: parent.right
                        y: Config.options.bar.bottom ? content.y - size : content.y + content.height
                    }
                }
            }
        }
    }
}

In the original

modules/ii/bar/Bar.qml is the same idea with more options: a "show bar when pressing Super" delay, a "push windows" option for auto-hide, dead-pixel workarounds, per-screen lists, and the corners implemented in BarContent. A separate VerticalBar exists for a vertical layout.

Module 2: The layout#

impulse/modules/bar/BarContent.qmlimport Quickshell
import QtQuick
import QtQuick.Layouts
import qs.services
import qs.modules.common
import qs.modules.common.widgets

// The bar's layout, like end-4: on the left the sidebar button and active window; in the middle
// three rounded groups (resources + media | workspaces | clock + utilities + battery);
// on the right the status icons (they open the right sidebar) and the tray.
// Scrolling on the left half changes brightness, on the right half volume.
Rectangle {
    id: root

    required property ShellScreen screen

    color: Appearance.colors.colLayer0

    MouseArea {
        anchors.fill: parent
        acceptedButtons: Qt.NoButton
        onWheel: event => {
            const step = event.angleDelta.y > 0 ? 0.05 : -0.05;
            if (event.x < width / 2) Brightness.set(Brightness.value + step);
            else Audio.setVolume(Audio.volume + step);
        }
    }

    // ---- left: sidebar button + active window ----
    RowLayout {
        anchors.left: parent.left
        anchors.leftMargin: 8
        anchors.verticalCenter: parent.verticalCenter
        spacing: 10

        RippleButton {
            id: leftButton
            implicitWidth: 32
            implicitHeight: 32
            buttonRadius: Appearance.rounding.full
            toggled: GlobalStates.sidebarLeftOpen
            onClicked: GlobalStates.sidebarLeftOpen = !GlobalStates.sidebarLeftOpen
            contentItem: MaterialSymbol {
                horizontalAlignment: Text.AlignHCenter
                text: "auto_awesome"
                iconSize: Appearance.font.pixelSize.larger
                color: leftButton.toggled ? Appearance.colors.colOnPrimary : Appearance.colors.colOnLayer0
            }
        }
        ActiveWindow {
            Layout.maximumWidth: root.width / 2 - centerGroups.width / 2 - 70
        }
    }

    // ---- centre: three groups ----
    RowLayout {
        id: centerGroups
        anchors.centerIn: parent
        spacing: 4

        BarGroup {
            Resources {}
            Media { Layout.maximumWidth: 200 }
        }
        BarGroup {
            Workspaces { screen: root.screen }
        }
        BarGroup {
            ClockWidget {}
            UtilButtons {}
            BatteryIndicator {}
        }
    }

    // ---- right: indicators + tray ----
    RowLayout {
        anchors.right: parent.right
        anchors.rightMargin: 8
        anchors.verticalCenter: parent.verticalCenter
        spacing: 6

        SysTray {}

        RippleButton {
            implicitWidth: indicators.implicitWidth + 20
            implicitHeight: 32
            buttonRadius: Appearance.rounding.full
            toggled: GlobalStates.sidebarRightOpen
            onClicked: GlobalStates.sidebarRightOpen = !GlobalStates.sidebarRightOpen

            contentItem: RowLayout {
                id: indicators
                spacing: 8
                readonly property color fg: GlobalStates.sidebarRightOpen ? Appearance.colors.colOnPrimary : Appearance.colors.colOnLayer0

                MaterialSymbol { visible: Audio.muted; text: "volume_off"; color: indicators.fg; Layout.leftMargin: 10 }
                MaterialSymbol { visible: Audio.micMuted; text: "mic_off"; color: indicators.fg }
                StyledText {
                    visible: Notifications.unread > 0
                    text: Notifications.unread
                    color: indicators.fg
                    font.pixelSize: Appearance.font.pixelSize.smaller
                }
                MaterialSymbol { text: "notifications"; color: indicators.fg; Layout.leftMargin: Audio.muted || Audio.micMuted ? 0 : 10 }
                MaterialSymbol { text: "wifi"; color: indicators.fg }
                MaterialSymbol { text: "bluetooth"; color: indicators.fg; Layout.rightMargin: 10 }
            }
        }
    }
}

            fill: 1
            iconSize: 14
            color: Appearance.colors.colOnSecondaryContainer
        }
    }
    StyledText {
        Layout.fillWidth: true
        text: (root.player?.trackTitle ?? "") + (root.player?.trackArtist ? " • " + root.player.trackArtist : "")
        elide: Text.ElideRight
        color: Appearance.colors.colOnLayer1
    }

    TapHandler {
        acceptedButtons: Qt.LeftButton | Qt.MiddleButton | Qt.RightButton
        onTapped: (eventPoint, button) => {
            if (button === Qt.MiddleButton) root.player.next();
            else if (button === Qt.RightButton) root.player.previous();
            else root.player.togglePlaying();
        }
    }
}

A ring showing track progress around a play/pause icon, then "title • artist". A TapHandler (not a MouseArea, which would take up a slot in the layout) handles left click = play/pause, middle = next, right = previous.

With auto-hide on, show the bar while Super is held. (It's already in mustShow; check that the workspaceNumber keybind from extras/hyprland.conf is set, then explain why holding Super reveals the bar.)

✅ Solution

mustShow includes GlobalStates.superDown, which the workspaceNumber global shortcut sets on press and clears on release. Auto-hide checks mustShow, so the bar slides in while Super is held.
