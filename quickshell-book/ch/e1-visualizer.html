e1Eye Candy 1: The Audio Visualizer

Bars that jump with the music are the most requested widget in ricing. The audio comes from cava, a small program that listens to PipeWire and turns the sound into a spectrum. This chapter runs cava inside the shell and draws three visualizers from its numbers: bars, a smooth wave, and a ring to put around album art.

Getting the spectrum#

cava usually draws bars in a terminal. With raw output and the ascii format it prints one line per frame instead, each line a list of numbers like 12;40;100;67;…. The shell reads those lines.

eyecandy/services/Cava.qmlpragma Singleton
import QtQuick
import Quickshell
import Quickshell.Io

// The audio spectrum, from cava. cava normally draws bars in a terminal; with
// "raw" output and the "ascii" format it prints one line per frame instead:
// numbers from 0 to 100 separated by ";", one per bar. We read those lines.
//
// cava only runs while something uses it: a widget calls acquire() when it
// appears and release() when it goes, so a hidden visualizer costs nothing.
Singleton {
    id: root

    property int bars: 32
    property int framerate: 60
    readonly property bool available: !missing
    property bool missing: false
    property var values: new Array(bars).fill(0)      // 0..1 per bar
    property int users: 0
    readonly property bool idle: values.every(v => v === 0)

    function acquire() { users++; }
    function release() { users = Math.max(0, users - 1); }

    // cava's config, generated, so the shell decides the bars and the rate
    readonly property string config: `
[general]
bars = ${bars}
framerate = ${framerate}
sleep_timer = 1
autosens = 1

[input]
method = pipewire
source = auto

[output]
method = raw
raw_target = /dev/stdout
data_format = ascii
ascii_max_range = 100
bar_delimiter = 59

[smoothing]
monstercat = 1
noise_reduction = 70
`

    Process {
        id: proc
        running: root.users > 0 && !root.missing
        // write the config to a file (path and text as arguments), then run cava on it
        command: ["sh", "-c", 'command -v cava >/dev/null || exit 127; f="${XDG_RUNTIME_DIR:-/tmp}/qs-cava.conf"; printf "%s" "$1" > "$f" && exec cava -p "$f"', "sh", root.config]
        stdout: SplitParser {
            // one line = one frame: "12;40;100;…;"
            onRead: line => {
                const parts = line.split(";");
                const out = new Array(root.bars);
                for (let i = 0; i < root.bars; i++)
                    out[i] = Math.min(1, (Number(parts[i]) || 0) / 100);
                root.values = out;
            }
        }
        onExited: code => {
            if (code === 127) {
                root.missing = true;
                console.warn("[Cava] cava is not installed: sudo pacman -S cava");
            }
            root.values = new Array(root.bars).fill(0);
        }
    }
}

SplitParser calls onRead for every line cava prints. Split at ;, convert each number, scale 0–100 to 0–1, and publish the new array. At 60 frames a second, this is the only work per frame.

Exit code 127 means cava isn't installed: remember that and don't retry. In any case, drop the bars to zero.

A built-in alternative

Some Quickshell builds (the noctalia fork) have a PwAudioSpectrum type that computes the spectrum inside Quickshell. Upstream Quickshell 0.3 doesn't, so cava is the portable choice; lyne-dots and Caelestia use it too.

Bars#

eyecandy/widgets/VisualizerBars.qmlimport QtQuick
import qs.services

// Spectrum bars. Each bar's height follows one value from cava.
//   VisualizerBars { width: 300; height: 60; mirrored: true }
// mirrored: bars grow up and down from the middle (a "waveform" look)
// gradient: colours from the left bar to the right bar
Item {
    id: root

    property bool mirrored: false
    property real spacing: 3
    property color colorFrom: "#7dcfff"
    property color colorTo: "#bb9af7"
    property real minHeight: 2              // silent bars stay visible as dots
    readonly property int count: Cava.bars

    Component.onCompleted: Cava.acquire()
    Component.onDestruction: Cava.release()

    Repeater {
        model: root.count
        Rectangle {
            required property int index
            readonly property real value: Cava.values[index] ?? 0
            readonly property real barWidth: (root.width - root.spacing * (root.count - 1)) / root.count

            x: index * (barWidth + root.spacing)
            width: barWidth
            height: Math.max(root.minHeight, value * root.height)
            y: root.mirrored ? (root.height - height) / 2 : root.height - height
            radius: width / 2
            // blend the two colours by the bar's place in the row
            color: Qt.tint(root.colorFrom, Qt.alpha(root.colorTo, index / Math.max(1, root.count - 1)))
            // cava already smooths; a short animation hides the frame steps
            Behavior on height { NumberAnimation { duration: 50 } }
        }
    }
}

A smooth wave#

eyecandy/widgets/VisualizerWave.qmlimport QtQuick
import QtQuick.Shapes
import qs.services

// A smooth wave through the spectrum values, drawn as one filled shape.
// Each value is a point; curves between the points make it flow instead
// of looking like steps.
Item {
    id: root

    property color color: "#7aa2f7"
    property real opacityFill: 0.35

    Component.onCompleted: Cava.acquire()
    Component.onDestruction: Cava.release()

    // the points, recomputed every frame: x spread across the width, y from the value
    readonly property var points: {
        const v = Cava.values, n = v.length, pts = [];
        for (let i = 0; i < n; i++)
            pts.push(Qt.point(i / (n - 1) * width, height - (v[i] ?? 0) * height * 0.9));
        return pts;
    }

    // SVG path text: start at the bottom left, curve through every point, close at the bottom right.
    // Each curve's control points sit halfway between two points, which gives a smooth line.
    readonly property string path: {
        const p = points;
        if (p.length < 2)
            return "";
        let d = `M 0 ${height} L ${p[0].x} ${p[0].y}`;
        for (let i = 1; i < p.length; i++) {
            const mx = (p[i - 1].x + p[i].x) / 2;
            d += ` C ${mx} ${p[i - 1].y} ${mx} ${p[i].y} ${p[i].x} ${p[i].y}`;
        }
        return d + ` L ${width} ${height} Z`;
    }

    Shape {
        anchors.fill: parent
        preferredRendererType: Shape.CurveRenderer
        ShapePath {
            strokeColor: root.color
            strokeWidth: 2
            fillColor: Qt.alpha(root.color, root.opacityFill)
            PathSvg { path: root.path }
        }
    }
}

An SVG path string: start at the bottom left, rise to the first point, then a cubic curve to each next point. The trick for smoothness: both control points sit at the horizontal midpoint between two points, at the heights of each end. Curves then meet with matching slopes, so the line flows instead of zig-zagging. Finally down to the bottom right and close.

A ring#

eyecandy/widgets/VisualizerRing.qmlimport QtQuick
import qs.services

// Bars arranged in a circle, pointing outwards: put album art in the middle
// and the music pulses around it.
//   VisualizerRing { width: 300; height: 300; radius: 100 }
Item {
    id: root

    property real radius: Math.min(width, height) / 2 * 0.62   // inner circle where bars start
    property real maxLength: Math.min(width, height) / 2 - radius
    property real barWidth: 4
    property color colorFrom: "#7dcfff"
    property color colorTo: "#bb9af7"
    // cava's bars once around the circle, then mirrored, so the ring is symmetric
    readonly property int count: Cava.bars * 2

    Component.onCompleted: Cava.acquire()
    Component.onDestruction: Cava.release()

    Repeater {
        model: root.count
        Item {
            required property int index
            // which cava bar: 0 … bars-1, then back down, mirrored
            readonly property int source: index < Cava.bars ? index : root.count - 1 - index
            readonly property real value: Cava.values[source] ?? 0

            // an item at the centre, rotated to this bar's angle; the bar sits on it
            x: root.width / 2
            y: root.height / 2
            rotation: index / root.count * 360

            Rectangle {
                x: -root.barWidth / 2
                y: -root.radius - height               // outward from the inner circle
                width: root.barWidth
                height: Math.max(2, parent.value * root.maxLength)
                radius: width / 2
                color: Qt.tint(root.colorFrom, Qt.alpha(root.colorTo, parent.source / Math.max(1, Cava.bars - 1)))
                Behavior on height { NumberAnimation { duration: 50 } }
            }
        }
    }
}

Performance#

Visualizers redraw up to 60 times a second, so a few habits matter:

  • Run cava only while visible: the acquire/release pattern. Put visualizers on panels that unload when closed (LazyLoader), and they stop by themselves.
  • Fewer bars in small places: 32 is plenty for a bar, 64 at most for a big desktop widget.
  • Lower the frame rate on laptops: Cava.framerate = 30 halves the work and still looks fluid.
  • cava's sleep_timer already stops the frames during silence.
