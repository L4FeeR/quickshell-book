# Tokyo Night GRUB theme
# Pictures and fonts come from make-assets.sh (run it once before installing).
# Positions are in pixels, percentages, or both ("50%-200"): the theme scales
# with the screen. Colours are #rrggbb.

# ── The whole screen ────────────────────────────────────────────────────────
title-text: ""                     # GRUB's own title off: we draw our own
desktop-image: "background.png"
desktop-color: "#16161e"           # behind the image, and if it fails to load
terminal-font: "Tokyo Regular 16"  # the command line (press c)
terminal-box: "menu_*.png"
message-font: "Tokyo Regular 16"
message-color: "#c0caf5"

# ── Logo and title above the menu ──────────────────────────────────────────
+ image {
    left = 50%-36
    top = 20%-36
    file = "logo.png"
}
+ label {
    left = 0
    top = 20%+48
    width = 100%
    align = "center"
    text = "Choose a system"
    font = "Tokyo Bold 22"
    color = "#c0caf5"
}

# ── The menu ───────────────────────────────────────────────────────────────
+ boot_menu {
    left = 50%-360
    top = 32%
    width = 720
    height = 36%

    menu_pixmap_style = "menu_*.png"         # the panel behind the entries
    item_font = "Tokyo Regular 18"
    item_color = "#a9b1d6"
    selected_item_font = "Tokyo Bold 18"
    selected_item_color = "#1a1b26"
    selected_item_pixmap_style = "select_*.png"   # the highlight bar

    icon_width = 32
    icon_height = 32
    item_icon_space = 16                     # gap between icon and text
    item_height = 40
    item_padding = 8                         # inside each entry
    item_spacing = 30                        # between entries: the highlight bar is drawn around the entry and grows into this gap
    scrollbar = false
}

# ── The countdown, as a bar that empties ───────────────────────────────────
+ progress_bar {
    id = "__timeout__"                       # this id makes it the timeout
    left = 50%-360
    top = 76%
    width = 720
    height = 26
    bar_style = "progress_bar_*.png"
    highlight_style = "progress_highlight_*.png"
    font = "Tokyo Regular 14"
    text_color = "#c0caf5"
    text = "@TIMEOUT_NOTIFICATION_MIDDLE@"   # "The highlighted entry will boot in 5s"
}

# ── Key help at the bottom ─────────────────────────────────────────────────
+ label {
    left = 0
    top = 100%-44
    width = 100%
    align = "center"
    text = "@KEYMAP_MIDDLE@"                 # "↑ ↓ to select, Enter to boot, e to edit…"
    font = "Tokyo Regular 14"
    color = "#545c7e"
}
